\begin{frame}{Formalizing next-token prediction (NTP)}
  \begin{columns}[t]
    \begin{column}{0.48\linewidth}
      \begin{block}{Autoregressive factorization}
        By the chain rule, the joint probability of $X=(x_1,\dots,x_T)$ factorizes exactly into conditional steps:
        \[
          P(X) = \prod_{t=1}^{T} P(x_t \mid x_1, \dots, x_{t-1})
        \]
        Modeling a text sequence is equivalent to predicting the next token from every prefix $x_{<t}$.
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{block}{Self-supervised training loss}
        Natural text is self-labeled: each token $x_t$ is the ground-truth target for prefix $x_{<t}$. We minimize cross-entropy:
        \[
          \mathcal{L}(\theta) = -\frac{1}{T}\sum_{t=1}^{T} \log P_\theta(x_t \mid x_{<t})
        \]
        One sequence of length $T$ provides $T$ supervised prediction targets at once.
      \end{block}
    \end{column}
  \end{columns}
  \smallsource{Bengio et al. (2003); Radford et al. (2018, 2019)}
\end{frame}

\section{Attention from first principles}

\begin{frame}{Step 1: Tokens become vectors in a shared space}
  \begin{columns}[c]
    \begin{column}{0.55\linewidth}
      \begin{itemize}
          \setlength{\itemsep}{7pt}
        \item Input token IDs are integers in $\{0,\dots,V-1\}$ with shape $B\times T$.
        \item An embedding lookup table $W_E\in\mathbb{R}^{V\times d}$ maps each token ID to a vector of width $d$:
      \end{itemize}
      \[
        X = W_E[\text{ids}] \quad\in\mathbb{R}^{B\times T\times d}
      \]
      \begin{alertblock}{Isolated representations}
        At this stage, each row $x_t \in \mathbb{R}^d$ only knows its own token identity---it knows nothing about surrounding tokens.
      \end{alertblock}
    \end{column}
    \begin{column}{0.41\linewidth}
      \centering
      \begin{tikzpicture}[node distance=0.38cm, every node/.style={font=\small}]
        \node[draw=themebluedraw,fill=themebluebg,rounded corners,minimum width=3.2cm,minimum height=.65cm] (ids) {Token IDs ($B\times T$)};
        \node[draw=themebluedraw,fill=themebluecard,rounded corners,below=of ids,minimum width=3.2cm,minimum height=.65cm] (we) {Lookup table $W_E$};
        \node[draw=successdraw,fill=successbg,rounded corners,below=of we,minimum width=3.2cm,minimum height=.65cm] (x) {Token vectors $X$ ($B\times T\times d$)};
        \draw[-{Stealth},thick] (ids)--(we);
        \draw[-{Stealth},thick] (we)--(x);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

  \vfill
  \begin{block}{The goal of self-attention}
    For every token $i$, compute a relevance weight $\alpha_{i,j} \ge 0$ ($\sum_{j\le i}\alpha_{i,j}=1$) for each allowed token $j \le i$, and form a weighted mixture of their information.
  \end{block}
  \smallsource{Bahdanau et al. (2014); Vaswani et al. (2017)}
\end{frame}

\begin{frame}{Step 3: Queries, Keys, and Values}
  \small Each token vector $x_t$ is projected into three distinct vectors via learned matrices $W_Q, W_K, W_V$:
  \par\vspace{4pt}
  \centering
  \begin{tikzpicture}[
      x=1.0cm,
      every node/.style={font=\footnotesize,align=center},
      rolebox/.style={draw=themebluedraw,fill=themebluebg,rounded corners,text width=3.1cm,minimum height=1.22cm}
    ]
    \node[rolebox] at (0,0) {\textbf{Query ($q_t = x_t W_Q$)}\\[2pt]\textcolor{neutraldark}{What context is this token looking for?}};
    \node[rolebox] at (3.65,0) {\textbf{Key ($k_t = x_t W_K$)}\\[2pt]\textcolor{neutraldark}{What features does each token offer?}};
    \node[rolebox,draw=successdraw,fill=successbg] at (7.3,0) {\textbf{Value ($v_t = x_t W_V$)}\\[2pt]\textcolor{neutraldark}{What content moves when selected?}};
  \end{tikzpicture}
  \par\vspace{4pt}
  \begin{itemize}
      \setlength{\itemsep}{3pt}
    \item \textbf{Matching}: Compare token $i$'s query $q_i$ with token $j$'s key $k_j$ via dot product $s_{i,j} = q_i^\top k_j$.
    \item \textbf{Retrieving}: Normalize scores with softmax into weights $\alpha_{i,j}$ and mix values: $\text{output}_i = \sum_{j \le i} \alpha_{i,j} v_j$.
  \end{itemize}
  \begin{alertblock}{Why three projections instead of using $x_t$ directly?}
    \small Asymmetry: what token $i$ searches for ($q_i$) differs from the features it offers ($k_i$) and the content it transmits ($v_i$).
  \end{alertblock}
\end{frame}

\begin{frame}{Step 4: Computing attention for all tokens in parallel}
  Stacking queries, keys, and values into matrices $Q, K, V \in \mathbb{R}^{T\times d_h}$ computes all pairwise relationships in one formula:
  \[
    \operatorname{Attn}(Q,K,V)
    = \underbrace{\operatorname{softmax}\!\left(\overbrace{QK^\top}^{\text{1. similarity scores}} + \overbrace{M}^{\text{2. mask}}\right)}_{\text{3. attention weights } A \in \mathbb{R}^{T\times T}} \underbrace{V}_{\text{4. value mix}}
  \]
  \centering
  \begin{tikzpicture}[
      x=0.92cm,
      every node/.style={font=\scriptsize,align=center},
      stepbox/.style={draw=themebluedraw,fill=themebluebg,rounded corners,text width=2.15cm,minimum height=1.15cm}
    ]
    \node[stepbox] (s1) at (0,0) {\textbf{1. Score pairs}\\$S = QK^\top$\\[2pt]\textcolor{inkmuted}{all $(i,j)$ dot products}};
    \node[stepbox,draw=leafreddraw,fill=leafredbg] (s2) at (2.85,0) {\textbf{2. Causal mask}\\$S + M$\\[2pt]\textcolor{signaldark}{hide future $j > i$}};
    \node[stepbox,draw=accentdarkdraw,fill=accentdarkbg] (s3) at (5.7,0) {\textbf{3. Softmax}\\$A = \operatorname{softmax}(\cdot)$\\[2pt]\textcolor{accentdark}{row weights sum to $1$}};
    \node[stepbox,draw=successdraw,fill=successbg] (s4) at (8.55,0) {\textbf{4. Mix values}\\$O = AV$\\[2pt]\textcolor{successgreen}{contextualized states}};

    \draw[-{Stealth},thick,neutraldark] (s1)--(s2);
    \draw[-{Stealth},thick,neutraldark] (s2)--(s3);
    \draw[-{Stealth},thick,neutraldark] (s3)--(s4);
  \end{tikzpicture}
  \smallsource{Vaswani et al. (2017); Rush (2018, The Annotated Transformer) --- in code, divide scores by $\sqrt{d_h}$ to keep logit variance $\approx 1$}
\end{frame}

\begin{frame}{Causality is a constraint on logits}
  \begin{columns}[c]

            \ifnum\pgfmathresult=1
            \fill[themebluecard] (\j,-\i) rectangle +(0.9,-0.9);
            \else
            \fill[leafredbg] (\j,-\i) rectangle +(0.9,-0.9);
            \fi
            \draw[white] (\j,-\i) rectangle +(0.9,-0.9);
          }
        }
        \node[above,font=\scriptsize] at (2.45,.15) {key position $j$};
        \node[rotate=90,font=\scriptsize] at (-.55,-2.45) {query position $i$};
        \node[font=\scriptsize] at (2.45,-5.35) {\textcolor{themebluedraw}{\textbf{visible ($0$)}} \quad/\quad \textcolor{leafreddraw}{\textbf{blocked ($-\infty$)}}};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{itemize}
          \setlength{\itemsep}{4pt}
        \item Position $i$ may attend only to keys $j \le i$.
        \item Future scores ($j > i$) receive $-\infty$ \textbf{before} softmax so $\exp(-\infty)=0$.
        \item Zeroing probabilities after softmax breaks row normalization.
      \end{itemize}
      \begin{block}{PyTorch implementation tip}
        \small Register a lower-triangular mask via \texttt{register\_buffer} and apply \texttt{scores.masked\_fill(mask == 0, -inf)}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Multi-head attention: multiple parallel retrieval subspaces}
  \begin{columns}[c]
    \begin{column}{0.52\linewidth}
      One token needs multiple types of context at once (syntax, coreference, local phrases). We split width $d$ into $H$ heads of size $d_h = d / H$:
      \begin{enumerate}
          \setlength{\itemsep}{4pt}
          \footnotesize
        \item \textbf{Project}: $Q, K, V \in \mathbb{R}^{B \times T \times d}$ via $W_Q, W_K, W_V \in \mathbb{R}^{d\times d}$.
        \item \textbf{Split heads}: \texttt{view(B, T, H, d\_h).transpose(1, 2)} $\to B \times H \times T \times d_h$.
        \item \textbf{Attend per head}: $S = QK^\top / \sqrt{d_h} \in \mathbb{R}^{B \times H \times T \times T}$, mix $O_h = AV_h$.
        \item \textbf{Merge \& project}: \texttt{transpose(1, 2).view(B, T, d)} then multiply by $W_O \in \mathbb{R}^{d\times d}$.
      \end{enumerate}
    \end{column}
    \begin{column}{0.44\linewidth}
      \centering
      \begin{tikzpicture}[node distance=0.28cm, every node/.style={font=\scriptsize,align=center}, box/.style={draw=themebluedraw,fill=themebluebg,rounded corners,minimum width=3.5cm,minimum height=.48cm}]
        \node[box] (in) {Input $X$ ($B \times T \times d$)};
        \node[box,below=of in] (qkv) {Linear $W_Q, W_K, W_V$ + reshape\\($B \times H \times T \times d_h$)};
        \node[box,draw=leafreddraw,fill=leafredbg,below=of qkv] (attn) {$H$ parallel causal attention heads\\scores: $B \times H \times T \times T$};
        \node[box,draw=successdraw,fill=successbg,below=of attn] (out) {Concat heads + output proj $W_O$\\($B \times T \times d$)};
        \draw[-{Stealth},thick] (in)--(qkv);
        \draw[-{Stealth},thick] (qkv)--(attn);
        \draw[-{Stealth},thick] (attn)--(out);
      \end{tikzpicture}
      \vspace{2pt}
      \begin{alertblock}{Transpose trap}
        \footnotesize Always \texttt{transpose(1, 2)} between $T$ and $H$ so each head sees a full $T \times d_h$ sequence!
      \end{alertblock}
    \end{column}
  \end{columns}
  \smallsource{Vaswani et al. (2017)}
\end{frame}

\begin{frame}{Wait---attention alone is blind to token order!}
  \begin{columns}[c]
    \begin{column}{0.54\linewidth}
      \begin{alertblock}{Permutation problem}
        \small Dot products $q_i^\top k_j$ depend only on token vectors, not their indices. Without positions, attention treats the prefix like an unordered \textbf{bag of words}.
      \end{alertblock}
      \begin{block}{Solution: Positional Encoding (PE)}
        \small Inject position information so identical words at different positions have distinct vectors:
        \par\vspace{2pt}
        \centering $X_0 = W_E[\text{ids}] + P_{0:T} \quad\in\mathbb{R}^{B\times T\times d}$
        \par\vspace{2pt}\raggedright\scriptsize
        \textbf{GPT-2}: learned $W_{pe} \in \mathbb{R}^{T_{\max}\times d}$. \textbf{Modern LLMs}: RoPE rotates $Q, K$ by relative distance.
      \end{block}
    \end{column}
    \begin{column}{0.42\linewidth}
      \centering
      \begin{tikzpicture}[node distance=0.34cm, every node/.style={font=\small}]
        \node[draw=themebluedraw,fill=themebluebg,rounded corners,minimum width=3.2cm,minimum height=.60cm] (ids) {Token IDs ($B\times T$)};
        \node[draw=themebluedraw,fill=themebluecard,rounded corners,below=of ids,minimum width=3.2cm,minimum height=.60cm] (tok) {Token vectors $W_E[\text{ids}]$};
        \node[draw=leafreddraw,fill=leafredbg,rounded corners,below=of tok,minimum width=3.2cm,minimum height=.60cm] (pos) {$+$ Position vectors $P_{0:T}$};

  \end{columns}

\end{frame}

\section{The Transformer block}

\begin{frame}{A pre-norm Transformer block has two residual updates}
  \begin{columns}[c]
    \begin{column}{0.52\linewidth}
      \[
        \begin{aligned}
          X'  &= X + \operatorname{MHA}(\operatorname{Norm}_1(X)) \\
          X'' &= X' + \operatorname{MLP}(\operatorname{Norm}_2(X'))
        \end{aligned}
      \]
      \begin{itemize}
          \setlength{\itemsep}{5pt}
        \item \textbf{Self-attention} routes information \textbf{across positions} (sequence axis $T$).
        \item \textbf{The MLP} transforms features \textbf{within each position} independently (channel axis $d$).
        \item \textbf{Residual highway} ($X + \dots$) lets gradients flow unimpeded to early layers.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\linewidth}
      \centering
      \begin{tikzpicture}[node distance=.32cm,every node/.style={font=\scriptsize},box/.style={draw=themebluedraw,fill=themebluebg,rounded corners,minimum width=3.3cm,minimum height=.48cm}]
        \node[box] (x) {$X \in \mathbb{R}^{B \times T \times d}$};
        \node[box,below=of x] (n1) {$\operatorname{Norm}_1 \rightarrow$ Causal MHA};
        \node[box,fill=successbg,draw=successdraw,below=of n1] (a) {$X' = X + \text{attn\_out}$};
        \node[box,below=of a] (n2) {$\operatorname{Norm}_2 \rightarrow$ Per-token MLP};
        \node[box,fill=successbg,draw=successdraw,below=of n2] (b) {$X'' = X' + \text{mlp\_out}$};
        \draw[-{Stealth},thick] (x)--(n1);
        \draw[-{Stealth},thick] (n1)--(a);
        \draw[-{Stealth},thick] (a)--(n2);
        \draw[-{Stealth},thick] (n2)--(b);
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \smallsource{He et al. (2016, ResNet); Radford et al. (2019); Xiong et al. (2020)}
\end{frame}

\begin{frame}{Inside the MLP and the $1/3$ vs.\ $2/3$ parameter rule}
  \begin{columns}[c]
    \begin{column}{0.52\linewidth}
      \small The MLP applies a two-layer feed-forward network to each token vector $x_t \in \mathbb{R}^d$ independently:
      \[
        \operatorname{MLP}(x_t) = \operatorname{GELU}(x_t W_1 + b_1) W_2 + b_2
      \]
      \begin{itemize}
          \setlength{\itemsep}{4pt}
          \footnotesize
        \item \textbf{Expand}: $W_1 \in \mathbb{R}^{d \times 4d}$ projects $d \to 4d$.
        \item \textbf{Activate}: elementwise $\operatorname{GELU}$ non-linearity.
        \item \textbf{Contract}: $W_2 \in \mathbb{R}^{4d \times d}$ projects $4d \to d$.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\linewidth}
      \begin{block}{Parameter budget per block}
        \footnotesize
        \begin{itemize}
            \setlength{\itemsep}{2pt}
          \item \textbf{MHA} ($W_Q, W_K, W_V, W_O$): $\mathbf{4d^2}$ (\textbf{1/3})
          \item \textbf{MLP} ($W_1, W_2$): $\mathbf{8d^2}$ (\textbf{2/3})
          \item \textbf{Total per block}: $\mathbf{12d^2}$ parameters
        \end{itemize}
      \end{block}
      \begin{alertblock}{Takeaway for scaling}
        \footnotesize Two-thirds of block parameters and matrix-multiply FLOPs live in the MLP!
      \end{alertblock}
    \end{column}
  \end{columns}
  \smallsource{Vaswani et al. (2017); Hendrycks \& Gimpel (2016, GELU)}
\end{frame}

\begin{frame}{Normalization: stabilizing activations and gradients}
  \begin{columns}[t]
    \begin{column}{0.48\linewidth}
      \begin{block}{LayerNorm (per token across $d$)}
        For each token vector $x \in \mathbb{R}^d$:
        \[
          \operatorname{LN}(x) = \frac{x - \mu}{\sqrt{\sigma^2 + \epsilon}} \odot \gamma + \beta
        \]
        Centers to zero mean ($\mu$), scales to unit variance ($\sigma^2$), then applies learned $\gamma, \beta \in \mathbb{R}^d$.
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{alertblock}{Pre-Norm vs.\ Post-Norm}
        \begin{itemize}
            \setlength{\itemsep}{4pt}
            \footnotesize
          \item \textbf{Pre-Norm (GPT-2 / modern LLMs)}: $X + F(\operatorname{Norm}(X))$. Identity highway bypasses norm; stable gradients at initialization.
          \item \textbf{Post-Norm (2017 Transformer)}: $\operatorname{Norm}(X + F(X))$. Requires careful warmup; different model!
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \centering\small\textbf{Modern speedup:} \textbf{RMSNorm} drops mean-centering ($\mu$) and $\beta$, normalizing only by $\sqrt{\frac{1}{d}\sum_i x_i^2 + \epsilon}$.
  \smallsource{Ba et al. (2016, LayerNorm); Zhang \& Sennrich (2019, RMSNorm); Xiong et al. (2020)}
\end{frame}

\begin{frame}{Closing the loop: final norm, LM head, and target shift}
  \begin{columns}[c]
    \begin{column}{0.54\linewidth}
      After $N$ Transformer blocks, normalize $X_N$ and project to vocabulary logits:
      \[
        Z = \operatorname{Norm}_f(X_N)W_{\text{vocab}}^\top
        \quad\in\mathbb{R}^{B\times T\times V}
      \]
      \[
        \mathcal{L}=-\frac{1}{BT}\sum_{b,t}
        \log \operatorname{softmax}(Z_{b,t})_{y_{b,t}}
      \]
      \begin{itemize}
          \setlength{\itemsep}{4pt}
          \footnotesize
        \item \textbf{Weight tying (GPT-2)}: shares $W_{\text{vocab}} = W_E \in \mathbb{R}^{V \times d}$, saving $V \times d$ parameters ($\approx 38.6\text{M}$ in GPT-2 Small).
      \end{itemize}
    \end{column}
    \begin{column}{0.42\linewidth}
      \begin{block}{One stream, two views}
        \texttt{x = tokens[:, :-1]}\\[4pt]
        \texttt{y = tokens[:, 1:]}
      \end{block}
      \begin{alertblock}{Common bug}
        Forgetting the $+1$ shift trains the model to copy the current token ($x_t \to x_t$): loss drops to zero, generation fails completely.
      \end{alertblock}
    \end{column}
  \end{columns}
  \smallsource{Press \& Wolf (2017, Weight Tying); Radford et al. (2019)}
\end{frame}

\section{From GPT-2 to modern LLMs \& Lab 1}

\begin{frame}{Three Transformer families---and why Decoder-Only won}
  \begin{columns}[t]
    \begin{column}{0.31\linewidth}
      \begin{block}{Encoder-only}
        \footnotesize
        \textbf{BERT} (Devlin et al., 2018)\\[2pt]
        Bidirectional attention for classification and search. No autoregressive generation.
      \end{block}
    \end{column}
    \begin{column}{0.31\linewidth}
      \begin{block}{Encoder--decoder}
        \footnotesize
        \textbf{Original / T5} (Raffel, 2020)\\[2pt]
        Bidirectional source encoder + causal target decoder via cross-attention.
      \end{block}
    \end{column}
    \begin{column}{0.31\linewidth}
      \begin{block}{Decoder-only}
        \footnotesize
        \textbf{GPT / LLaMA} (Radford, 2019)\\[2pt]
        Single causal stream unifies all tasks as prompt completion + single KV cache.
      \end{block}
    \end{column}
  \end{columns}
  \begin{alertblock}{Why decoder-only dominates modern foundation models}
    \small Any task can be framed as prompt continuation, concentrating all parameters and pretraining compute into one unified causal stack with a single KV cache.
  \end{alertblock}
  \smallsource{Vaswani et al. (2017); Devlin et al. (2018); Radford et al. (2019); Brown et al. (2020)}
\end{frame}

\begin{frame}{What changed since GPT-2? The modern LLM stack}
  \small You will implement \textbf{GPT-2} in Lab 1. Modern LLMs keep the same macro-architecture while upgrading five components:
  \vspace{2pt}
  \centering\scriptsize
  \begin{tabular}{@{}>{\bfseries\raggedright\arraybackslash}p{1.85cm}>{\raggedright\arraybackslash}p{2.25cm}>{\raggedright\arraybackslash}p{2.85cm}>{\raggedright\arraybackslash}p{3.35cm}@{}}
    \toprule
    Component & GPT-2 (2019) & Modern LLM (2023+) & Primary rationale \\
    \midrule
    Positions & Learned $W_{pe}$ & \textbf{RoPE} (Su et al., 2021) & Relative distance, length extrapolation \\
    Normalization & Pre-\textbf{LayerNorm} & Pre-\textbf{RMSNorm} (Zhang, 2019) & Drops mean-centering; 10--50\% faster \\
    Attention & \textbf{MHA} ($H_Q = H_{KV}$) & \textbf{GQA} (Ainslie et al., 2023) & $4\times$--$8\times$ smaller KV cache in serving \\
    MLP activation & \textbf{GELU} ($d \to 4d \to d$) & \textbf{SwiGLU} (Shazeer, 2020) & Gated linear unit improves capacity \\
    Biases \& tying & Biases; tied $W_E$ & \textbf{Bias-free}; untied $W_{\text{vocab}}$ & Faster GEMMs; better stability at scale \\
    \bottomrule
  \end{tabular}
  \vfill
  \smallsource{Touvron et al. (2023, LLaMA); Jiang et al. (2023, Mistral); Raschka (LLM Architecture Gallery)}
\end{frame}
