\section{From documents to tokens}

\begin{frame}{Start with documents, not model inputs}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{block}{A document}
        A piece of text plus its source and identity: a book, article, web page, or other record.
      \end{block}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{Extract usable text}
        Keep content; remove or inspect navigation, markup, duplicated page furniture, and extraction errors.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{alertblock}{Where might web text come from?}
    Web archives store captured pages and related records (WARC); extracted-text files (WET) provide a text-oriented starting point. Neither is automatically clean.
  \end{alertblock}
\end{frame}

\begin{frame}{Filter, deduplicate, then split by document}
  \small
  \begin{enumerate}
      \setlength{\itemsep}{3pt}
    \item \textbf{Filter:} inspect rules for empty, short, or repetitive text; record what they remove.
    \item \textbf{Deduplicate:} group repeated documents before splitting, so no text sits in both train and validation.
    \item \textbf{Split:} assign documents to train or validation, then keep their streams separate.
  \end{enumerate}
  \begin{block}{Exact deduplication: one deterministic rule}
    Key each document by its whitespace-normalized text. In a fixed order, keep the first document of each key in its split, drop every later copy, and record (dropped ID, kept ID).
  \end{block}
  \textbf{Fit only on training data:} learn tokenizer merges on the training split, after deduplication; apply the tokenizer to both splits.
\end{frame}

\begin{frame}{Granularity moves cost between vocabulary and sequence}
  \small
  \begin{tabular}{llll}
    \toprule
    Unit & Vocabulary & Sequence & Main cost \\
    \midrule
    bytes & 256 & very long & attention and context \\
    words & millions & short & embeddings and unknowns \\
    subwords & tunable & moderate & learned merge system \\
    \bottomrule
  \end{tabular}
  \vfill
  \begin{block}{Why byte-level BPE?}
    Every UTF-8 string decomposes into 256 byte values, so the base system has no unknown text. Frequent byte sequences can then become single tokens.
  \end{block}
  \begin{alertblock}{System consequence}
    Tokenization changes sequence length, attention work, KV-cache size, multilingual efficiency, and the number of embedding parameters.
  \end{alertblock}
\end{frame}

\begin{frame}{BPE learns a greedy compression hierarchy}
  \begin{enumerate}
      \setlength{\itemsep}{6pt}
    \item Start from byte IDs and reserved special tokens.
    \item Count adjacent token pairs in the training corpus.
    \item Merge the most frequent pair into one new token.
    \item Repeat until the vocabulary reaches its target size.
  \end{enumerate}
  \[
    (p_0,p_1)\mapsto v_{\text{new}},
    \qquad
    \operatorname{repr}(v_{\text{new}})=
    \operatorname{repr}(p_0)\Vert\operatorname{repr}(p_1)
  \]
  \begin{block}{Encoding new text}
    Apply learned merges by rank. Earlier learned pairs win when several merges are possible.
  \end{block}
  \smallsource{Gage, Sennrich et al., GPT-2, and minBPE}
\end{frame}

\begin{frame}{A hand-worked merge exposes the algorithm}
  \small
  Corpus: \texttt{low low lower}. The space is a byte like any other, shown as \texttt{\textvisiblespace}.
  \vspace{2pt}
  \begin{center}
    \begin{tabular}{@{}llll@{}}
      \toprule
      Round & top counts & merged pair & stream \\
      \midrule
      0 & -- & -- & \texttt{l o w \textvisiblespace\ l o w \textvisiblespace\ l o w e r} \\
      1 & \texttt{(l,o)}:3\ \ \texttt{(o,w)}:3 & \texttt{(l,o)} $\to$ 257 & \texttt{lo w \textvisiblespace\ lo w \textvisiblespace\ lo w e r} \\
      2 & \texttt{(lo,w)}:3 & \texttt{(lo,w)} $\to$ 258 & \texttt{low \textvisiblespace\ low \textvisiblespace\ low e r} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{alertblock}{The rules that make training deterministic}
    \textbf{Ties:} the pair that appears first in the stream wins, so \texttt{(l,o)} beats \texttt{(o,w)}.
    \textbf{IDs:} bytes are 0--255, end-of-text is 256, merges start at 257.
    \textbf{Stop:} at the target vocabulary size, or when no pair occurs twice.
  \end{alertblock}
\end{frame}

\begin{frame}{The ``SolidGoldMagikarp'' story}
  \small
  \begin{enumerate}
      \setlength{\itemsep}{4pt}
    \item \textbf{Tokenizer data:} a Reddit username appeared repeatedly in counting threads. BPE learned \texttt{SolidGoldMagikarp} (with a leading space) as one token.
    \item \textbf{Pretraining data:} filtering can remove repetitive content from the data used to train the model, while the tokenizer keeps its vocabulary.
    \item \textbf{The mismatch:} a token frequent enough to be learned by BPE may be rare or absent during model training, leaving the model with a poorly learned representation.
  \end{enumerate}
  \begin{alertblock}{Pipeline lesson}
    Cleaning model data alone does not clean the tokenizer vocabulary. Compare the datasets and check token exposure.
  \end{alertblock}
  \footnotesize The exact cause is unconfirmed: the model's pretraining data is not public.
  \smallsource{\href{https://www.alignmentforum.org/posts/aPeJE8bSo6rAFoLqg}{Rumbelow and Watkins (2023)}; \href{https://aclanthology.org/2024.emnlp-main.649/}{Land and Bartolo (2024)}}
\end{frame}

\end{frame}

\begin{frame}{Bits per byte: compare on the same text}
  \small
  \begin{block}{Two different questions}
    \textbf{Tokenizer compression:} how many tokens represent the text?\\[3pt]
    \textbf{Model performance:} how much probability does the model assign to it?
  \end{block}
  For the same held-out text, express negative log-likelihood in \emph{bits per byte}:
  \[
    \text{bits / byte} =
    \frac{\text{total negative log-likelihood (nats)}}{\ln 2\;\times\;\text{UTF-8 bytes}}
    = \frac{\text{bits}}{\text{token}}\times\frac{\text{tokens}}{\text{byte}}.
  \]
  \begin{exampleblock}{Example}
    1,000 bytes, 400 tokens, summed negative log-likelihood 800 nats $\Rightarrow$ $1.15$ bits / byte.
  \end{exampleblock}
  Compare on identical text and boundary policy. Lower is better; fewer tokens alone do not guarantee a lower model loss.
\end{frame}

\checkpoint{tokenizer choice}
{A new tokenizer shortens a representative corpus by 20\% but adds 300 million embedding parameters.
Which metrics on both training and inference would decide whether this is a good trade?}

\begin{frame}{Tokenizer choice: one possible answer}
  \small
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item \textbf{Quality:} bits per byte on the same held-out text and boundary policy. Loss per token is not comparable across tokenizers.
    \item \textbf{Training:} 20\% fewer tokens for the same text means fewer positions and less attention work per document. Against that: 300M more embedding parameters to store and update, and rarer tail tokens.
    \item \textbf{Inference:} fewer decode steps and a smaller KV cache for the same text. Against that: larger weights and a more expensive vocabulary softmax at every step.
    \item \textbf{Decision:} a good trade if bits per byte is no worse at an equal training budget and the measured cost per byte of text falls. Check compression on the languages and domains that matter.
  \end{itemize}
\end{frame}

\section{Packing documents into examples}

\begin{frame}{Padding leaves unused positions}
  \small
  Suppose each example is padded to a 2,048-token window:

    \end{tikzpicture}
  \end{center}
  \begin{block}{Packing}
    Concatenate documents with explicit boundaries and slice fixed-length windows. The padding saved is set by the document-length distribution.
  \end{block}
\end{frame}

\begin{frame}{Packing improves utilization but changes attention structure}
  \centerline{\texttt{[doc A][EOT][doc B][EOT][doc C][EOT]}}
  \vspace{6pt}
  \begin{columns}[c]
    \begin{column}{0.56\linewidth}
      \begin{itemize}
          \setlength{\itemsep}{7pt}
        \item Every non-masked target can contribute to the loss.
        \item Windows can cross a document boundary.
        \item Standard causal attention lets doc B see doc A.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\linewidth}
      \begin{block}{Two valid policies}
        Accept cross-document context and record it.\\[6pt]
        Or use block-diagonal causal masks with document offsets.
      \end{block}
    \end{column}
  \end{columns}
  \begin{alertblock}{Invariant}
    Train and validation splits must happen at the document level before packing.
  \end{alertblock}
\end{frame}

\begin{frame}{The end-of-text token has two roles}
  \begin{enumerate}
      \setlength{\itemsep}{9pt}
    \item It preserves a recoverable boundary in a contiguous stream.
    \item It teaches the model that one document ended and another may begin.
  \end{enumerate}
  \begin{block}{Placement and count}
    Append it after \emph{every} document, the last one included: a stream of documents with $n_1,\dots,n_D$ tokens holds $\sum_j (n_j+1)$ tokens, and the next shard starts on a clean boundary.
  \end{block}
  \begin{alertblock}{Special-token policy}
    Reserve its ID before training merges. Prevent ordinary text from being confused with the special marker. Test round-trip behavior explicitly.
  \end{alertblock}
\end{frame}

\section{Storage and correctness}

\begin{frame}{One contiguous slice creates inputs and targets}
  \small
  Tokens are loaded from a flat 1D map $s$ of $n$ tokens.
  To form an input--target pair of context length $T$, take $T+1$ contiguous elements starting at offset $i$:
  \[
    x = s[i:i+T]
    \qquad
    y = s[i+1:i+T+1]
  \]
  \begin{center}
    \begin{tabular}{c|ccccc}
      & 0 & 1 & 2 & $\cdots$ & $T-1$ \\
      \hline
      $x$ & $s_i$ & $s_{i+1}$ & $s_{i+2}$ & $\cdots$ & $s_{i+T-1}$ \\
      $y$ & $s_{i+1}$ & $s_{i+2}$ & $s_{i+3}$ & $\cdots$ & $s_{i+T}$ \\
    \end{tabular}
  \end{center}
  \vfill
  \begin{block}{Valid offsets and a fast check}
    The target reads $s_{i+T}$, so $0\le i\le n-T-1$: sample $i$ uniformly from these $n-T$ offsets. Stack $B$ slices into $x,y$ of shape $(B,T)$, as 64-bit integers. For every batch, $y[:,:-1]$ must equal $x[:,1:]$ exactly.
  \end{block}
\end{frame}

\begin{frame}{A data pipeline needs end-to-end evidence}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Encode and decode arbitrary Unicode text losslessly.
    \item Verify deterministic token IDs and special-token handling.
    \item Measure compression on representative languages and domains.
    \item Audit document-level split membership before packing.
    \item Measure loader throughput; check for GPU input stalls when a trace is available.
  \end{enumerate}
  \begin{alertblock}{Do not optimize a corrupt stream}
    Decode sampled windows to inspect boilerplate and boundaries; compare document identities across splits to check leakage.
  \end{alertblock}
\end{frame}

\begin{frame}{Data bugs rarely crash loudly: know their symptoms}
  \small
  \begin{center}
    \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.40\linewidth}>{\raggedright\arraybackslash}p{.54\linewidth}@{}}
      \toprule
      Bug & Symptom \\
      \midrule
      No shift: $y=s[i:i+T]$ & The model copies its input: train and validation loss collapse, generation is useless \\ \addlinespace
      Offset allowed up to $i=n-T$ & A rare target window one token short: a shape error on some batches only \\ \addlinespace
      No end-of-text after the last document & Token count off by one; the next shard's first document is glued to this one \\ \addlinespace
      A validation copy of a training document & Validation loss lower than it should be, by an amount set by the overlap \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{alertblock}{Test invariants, not the loss curve}
    Shift equality, offset range, token count, disjoint splits: one assertion each.
  \end{alertblock}
\end{frame}

\checkpoint{pipeline audit}
{A training run is fast, but validation loss is implausibly low. List three document- or token-level checks that distinguish a good model from split contamination.}

\begin{frame}{Pipeline audit: one possible answer}
  \begin{itemize}
      \setlength{\itemsep}{9pt}
    \item \textbf{Document identities:} compare document IDs or content hashes across the train and validation splits. Any shared or duplicated document is contamination.
    \item \textbf{Order of operations:} confirm that deduplication and the document-level split happened before packing, and that merges were fitted on training text only.
    \item \textbf{Decoded windows:} decode sampled validation windows and search for them, or their boilerplate, in the training shards.
    \item Also rule out a target bug: $y[:, :-1]$ must equal $x[:, 1:]$.
  \end{itemize}
\end{frame}

\takeaways
{Byte-level BPE guarantees coverage, while merges trade vocabulary size for shorter sequences.}
{Pre-tokenization and special-token rules are part of the learned tokenizer.}
{Packing removes padding waste but requires an explicit document-boundary policy.}
{Binary shards are useful only when their schema, split, and target shift are verified.}
