  \end{center}
  \begin{block}{Inside one Transformer block}
    Linear projections and MLP $\rightarrow$ matrix multiplications; attention $\rightarrow$ matrix products, masking and softmax; LayerNorm, GELU and residuals $\rightarrow$ reductions and elementwise work.
  \end{block}
  \small The same model and batch can run on CPU or GPU; compare the work and its execution, not just device names.

\begin{frame}{CPU and GPU: different ways to execute work}
  \small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.18\linewidth}>{\raggedright\arraybackslash}p{.35\linewidth}>{\raggedright\arraybackslash}p{.35\linewidth}@{}}
    \toprule
    & CPU & GPU \\
    \midrule
    Design & Fewer complex cores; strong per-thread latency & Many execution lanes; high throughput on parallel work \\
    Good fit & Python, I/O, control flow; small or irregular work & Large batches of regular tensor operations \\
    Memory & Host memory and caches & Device memory and on-chip caches \\
    Trade-off & Less transfer/launch overhead for small work & Launch, transfer and synchronization costs \\
    \bottomrule
  \end{tabular}
  \vfill
  \begin{alertblock}{Measure the actual workload}
    A GPU can lose on small shapes or transfer-heavy work. Report batch, dtype, threads, device, and what is timed.
  \end{alertblock}
\end{frame}

\begin{frame}{What happens when PyTorch runs on CUDA?}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item CPU prepares a batch and issues PyTorch operations.
    \item Tensors are copied to the device when needed; GPU-resident weights and activations stay in device memory.
    \item The CPU enqueues GPU kernels; the GPU executes them asynchronously.
    \item Backward launches more kernels; the optimizer updates weights.
  \end{enumerate}
  \begin{alertblock}{Timing trap}
    A Python timer around an asynchronous launch can stop before the GPU finishes. Synchronize at timing boundaries or use CUDA events.
  \end{alertblock}
\end{frame}

\section{The machine beneath the tensor}

\begin{frame}{A GPU is a throughput machine with many lanes}
  \begin{columns}[c]
    \begin{column}{0.54\linewidth}
      \begin{itemize}
          \setlength{\itemsep}{7pt}
        \item A kernel launches a grid of threads.
        \item Threads execute in warps, commonly 32 lanes.
        \item Warps are scheduled onto streaming multiprocessors (SMs).
        \item Tensor cores accelerate supported matrix tiles.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\linewidth}
      \begin{tikzpicture}[node distance=.48cm,every node/.style={font=\scriptsize},box/.style={draw=themebluedraw,fill=themebluebg,rounded corners,minimum width=3.5cm,minimum height=.46cm}]
        \node[box] (grid) {kernel grid};
        \node[box,below=of grid] (block) {thread blocks};
        \node[box,fill=warningbg,draw=warningdraw,below=of block] (warp) {warps};
        \node[box,fill=successbg,draw=successdraw,below=of warp] (sm) {SM execution units};
        \draw[-{Stealth[length=4pt,width=3pt]},thick,draw=themenavy] (grid)--(block);
        \draw[-{Stealth[length=4pt,width=3pt]},thick,draw=themenavy] (block)--(warp);
        \draw[-{Stealth[length=4pt,width=3pt]},thick,draw=themenavy] (warp)--(sm);
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \smallsource{JAX Scaling Book, How to Think About GPUs}
\end{frame}

\begin{frame}{GEMM: the matrix multiplication behind our model}
  \begin{block}{General matrix--matrix multiplication}
    $C \leftarrow \alpha AB + \beta C$, with $A\in\mathbb{R}^{M\times K}$,
    $B\in\mathbb{R}^{K\times N}$ and $C\in\mathbb{R}^{M\times N}$.
    A plain matrix product is the case $\alpha=1,\ \beta=0$.
  \end{block}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{block}{In a Transformer}
        Linear layers reshape $B\times T$ token vectors into rows: Q/K/V, attention output, MLP and vocabulary projections.
      \end{block}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{In attention}
        Per-head $QK^\top$ forms scores; $PV$ mixes values. These are batched matrix products around a softmax.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small GEMMs dominate much of the model's arithmetic; shape, dtype and data movement determine speed.
\end{frame}

\begin{frame}{Tensor Cores and the rest of the GPU}
  \small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.23\linewidth}>{\raggedright\arraybackslash}p{.34\linewidth}>{\raggedright\arraybackslash}p{.31\linewidth}@{}}
    \toprule
    Unit & Typical work & Transformer examples \\
    \midrule
    Tensor Cores & Matrix-tile multiply--accumulate & Linear layers; attention products \\[5pt]
    CUDA cores & General scalar / vector arithmetic & Residual adds; optimizer updates \\[5pt]
    Other execution paths & Reductions and special functions & Norm statistics; softmax; GELU \\
    \bottomrule
  \end{tabular}
  \vfill
  \begin{alertblock}{These are typical mappings, not guarantees}
    Kernels combine work across units. Tensor Core use depends on GPU, dtype, shapes and backend; non-matmul operations and HBM traffic still matter.
  \end{alertblock}
  \smallsource{NVIDIA, Matrix Multiplication Background; GPU Performance Background}
\end{frame}

\begin{frame}{Warps reward regular work}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Good utilization}
        contiguous access\\similar control flow\\enough active warps\\tile-aligned matrix dimensions
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{alertblock}{Lost utilization}
        branch divergence\\strided loads\\register pressure\\tiny or awkward shapes
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \centering A mathematically identical operation can run much faster when its layout and dimensions match the hardware tiles.
\end{frame}

\begin{frame}{The memory hierarchy trades capacity for speed}
  \centering\small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.22\linewidth}>{\raggedright\arraybackslash}p{.16\linewidth}>{\raggedright\arraybackslash}p{.13\linewidth}>{\raggedright\arraybackslash}p{.36\linewidth}@{}}
    \toprule
    Level & Scope & Capacity & Role \\
    \midrule
    registers & one thread & tiny & live scalar and fragment state \\
    shared memory / SRAM & one block or SM & small & explicit reuse and tiling \\
    L2 cache & whole device & moderate & cross-SM reuse \\
    HBM / VRAM & whole device & large & weights, activations, optimizer state \\
    host memory & CPU system & largest & dataset staging and overflow \\
    \bottomrule
  \end{tabular}
  \vfill
  \begin{alertblock}{Performance principle}
    Moving a value often costs more time and energy than performing one arithmetic operation on it. Reuse data near the compute units.
  \end{alertblock}
\end{frame}

\begin{frame}{Map GPT operations to hardware questions}
  \small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.29\linewidth}>{\raggedright\arraybackslash}p{.30\linewidth}>{\raggedright\arraybackslash}p{.30\linewidth}@{}}
    \toprule
    In our training step & Typical work & Ask in the trace \\
    \midrule
    QKV and MLP & Matrix multiplications & Are shapes large enough? \\[4pt]
    Attention & Products and softmax & Which backend ran? \\[4pt]
    Norm, GELU, residual & Reductions and pointwise ops & Are launches or intermediates costly? \\[4pt]
    Data, loss, optimizer & Copies and updates & Is the GPU waiting? \\
    \bottomrule
  \end{tabular}
  \vfill
  Kernel choices depend on shape, dtype, hardware and implementation. Measure before drawing a conclusion.
\end{frame}

\section{Roofline reasoning}

\begin{frame}{Arithmetic intensity predicts the limiting resource}
  \begin{columns}[c]
    \begin{column}{0.69\linewidth}
      \centering
      \begin{tikzpicture}[x=.72cm,y=.72cm,every node/.style={font=\scriptsize}]
        \fill[themebluebg] (0,0)--(4.4,4.2)--(4.4,0)--cycle;
        \fill[warningbg] (4.4,0)--(4.4,4.2)--(9.2,4.2)--(9.2,0)--cycle;
        \draw[-{Stealth[length=4.5pt,width=3.2pt]},thick] (0,0)--(9.55,0);
        \draw[-{Stealth[length=4.5pt,width=3.2pt]},thick] (0,0)--(0,4.65) node[above,rotate=90,anchor=south east]{attained FLOP/s};
        \draw[very thick,themebluedraw] (0,0)--(4.4,4.2);
        \draw[very thick,warningdraw] (4.4,4.2)--(9.2,4.2);
        \draw[dashed,neutraldark] (4.4,0)--(4.4,4.2);
        \node[rotate=43,text=accentdark] at (2.35,2.65) {$I\,BW_{\mathrm{HBM}}$};
        \node[text=warningdraw] at (7.2,4.48) {$P_{\mathrm{peak}}$};
        \fill[themebluedraw] (1.15,1.10) circle (3pt);
        \node[anchor=west] at (1.28,.92) {vector add};
        \fill[warningdraw] (7.15,4.2) circle (3pt);
        \node[above] at (7.15,4.2) {large GEMM};
        \node[below] at (4.4,0) {ridge $P_{\mathrm{peak}}/BW$};
        \node[below,text=neutraldark] at (8.1,-.55) {arithmetic intensity $I$};
        \node[text=accentdark] at (2.1,.35) {bandwidth-bound};
        \node[text=warningdraw] at (6.9,.35) {compute-bound};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.29\linewidth}
      \scriptsize
      \[
        I=\frac{F}{B_{\mathrm{HBM}}}
      \]
      \[
        P\le\min(P_{\mathrm{peak}},I\,BW)
      \]
      \begin{alertblock}{Read the lower roof}
        Moving right means reusing each loaded byte more.
      \end{alertblock}
    \end{column}
  \end{columns}
  \smallsource{JAX Scaling Book, How to Think About GPUs}
\end{frame}

\begin{frame}{Vector addition: a bandwidth-bound model}
  For $C=A+B$ with FP32 elements:
  \[
    \text{FLOPs}=1,\qquad \text{bytes}=4+4+4=12
  \]
  \[
    I\approx 0.083\ \text{FLOP/byte}
  \]
  \begin{block}{Prediction}
    For a large vector under this traffic model, bandwidth can dominate. Small vectors can instead be launch-limited.
  \end{block}
  \begin{alertblock}{Benchmark check}
    Doubling peak FLOPs should barely change runtime if memory bandwidth stays fixed.
  \end{alertblock}
\end{frame}

\begin{frame}{Large matrix multiplication creates reuse}
  For square $C=AB$, matrices of width $n$:
  \[
    \text{FLOPs}\approx2n^3,\qquad \text{minimum bytes}\approx3n^2s
  \]
  \[
    I\approx\frac{2n}{3s}
  \]
  This idealized intensity assumes each matrix element is transferred once. Tiling increases reuse; actual traffic may be higher.
  \begin{alertblock}{Small matrices behave differently}
    Launch overhead, poor tile occupancy, and insufficient reuse can keep small GEMMs far below peak compute.
  \end{alertblock}
\end{frame}

\begin{frame}{Fusion: the same math, fewer intermediates}
  \small
  Consider an illustrative pointwise chain $y=\operatorname{ReLU}(x+b)+r$, with equal-sized tensors:
  \vspace{3pt}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{alertblock}{Separate kernels}
        $u=x+b$ \quad $\rightarrow$ write $u$\\[3pt]
        $v=\operatorname{ReLU}(u)$ \quad $\rightarrow$ write $v$\\[3pt]
        $y=v+r$ \quad $\rightarrow$ write $y$
      \end{alertblock}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{One fused kernel}
        Read $x,b,r$; compute the chain; write $y$.\\[5pt]
        $u$ and $v$ can stay in registers instead of being written and re-read.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  Idealized traffic: 8 versus 4 element transfers. Cache reuse, broadcasting and actual kernels change the result; measure the trace.
\end{frame}

\begin{frame}{Why is attention difficult to run fast?}
  \small
  \[
    S=QK^\top/\sqrt{d_h}\quad\longrightarrow\quad
    P=\operatorname{softmax}(S+\text{mask})\quad\longrightarrow\quad O=PV
  \]
  \begin{itemize}
      \setlength{\itemsep}{5pt}
    \item The two matrix products can use fast matrix hardware; masking and row-wise softmax add reductions and elementwise work.
    \item Writing and rereading $S$ and $P$, each $T\times T$ per head, costs quadratic storage and HBM traffic.
    \item Irregular tiles, causal masking and limited on-chip memory complicate efficient GPU execution.
  \end{itemize}
  \begin{alertblock}{The target}
    Preserve the attention result while moving fewer intermediates through HBM.
  \end{alertblock}
\end{frame}

\begin{frame}{FlashAttention: tile, fuse, and avoid the large intermediates}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{alertblock}{Straightforward attention}
        $QK^\top$ $\to$ write scores\\[3pt]
        softmax $\to$ write probabilities\\[3pt]
        $PV$ $\to$ write output
      \end{alertblock}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{FlashAttention}
        Load blocks of $Q,K,V$; compute score and value tiles; update online softmax statistics and partial outputs on chip.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small It avoids materializing the full $T\times T$ scores and probabilities in HBM; backward recomputes tiles using saved statistics. Dense attention still has quadratic arithmetic. Results can differ slightly from rounding.
  \smallsource{Dao et al., FlashAttention (2022)}
\end{frame}

\begin{frame}{Using efficient attention in this course}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Module 1 implements $QK^\top$, masking, softmax and $PV$ explicitly.
    \item In the GPU lab, switch to PyTorch \texttt{scaled\_dot\_product\_attention} (SDPA) and compare outputs and gradients.
    \item SDPA selects an eligible backend; it may use FlashAttention, another fused kernel, or the math implementation.
    \item Inspect the actual backend and profile the update: hardware, dtype, shapes and mask affect eligibility and performance.
  \end{enumerate}
  \smallsource{PyTorch SDPA documentation; Dao et al., FlashAttention (2022)}
\end{frame}

\section{Memory, compilation and profiling}

\begin{frame}{VRAM holds much more than model weights}
  \small
  \begin{tabular}{lll}
    \toprule
    Component & Typical scaling & Lifetime \\
    \midrule
    parameters & $P\times$ dtype bytes & whole run \\
    gradients & roughly $P$ & until update/reset \\
    Adam moments & roughly $2P$ in FP32 & whole run \\
    master weights & optional FP32 copy & whole run \\
    activations & $B,T,L,d$ dependent & forward to backward \\
    temporary buffers & kernel dependent & operation lifetime \\
    allocator reserve & fragmentation dependent & process lifetime \\
    \bottomrule
  \end{tabular}
  \vfill
  \begin{block}{Useful distinction}
    Report allocated and reserved memory. Out-of-memory behavior depends on peaks and fragmentation, not only steady-state tensor sizes.
  \end{block}
\end{frame}

\begin{frame}{Activation checkpointing exchanges compute for memory}
  \begin{columns}[c]
    \begin{column}{0.48\linewidth}
      \begin{alertblock}{Store everything}
        Faster backward pass\\higher activation memory\\larger feasible recomputation-free batch
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{block}{Checkpoint selected boundaries}
        Lower activation memory\\recompute forward segments in backward\\may enable larger $B$ or $T$
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  Evaluate end-to-end tokens per second. A larger batch does not guarantee a faster run if recomputation dominates.
\end{frame}

\begin{frame}{Performance metrics answer different questions}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Throughput}
        tokens/s measures delivered training work\\step time exposes update cadence\\time-to-target includes optimization quality
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{Hardware efficiency}
        \[\mathrm{MFU}=\frac{\text{model FLOPs/s}}{\text{peak FLOPs/s}}\]
        MFU depends on the FLOP model and quoted peak precision.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Mixed precision: a performance and numerical trade-off}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{block}{What changes?}
        Autocast runs supported operations in BF16 and retains higher precision where appropriate. The same training objective and update remain.
      \end{block}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{Why try it?}
        Supported hardware may run matrix operations faster and store some activations in fewer bytes. The gain depends on the workload.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{alertblock}{Measure before claiming a speedup}
    Check hardware support, loss and gradients; compare identical shapes and attention paths. FP16 loss scaling is a separate consideration.
  \end{alertblock}
\end{frame}

\begin{frame}{What does \texttt{torch.compile} do?}
  \begin{enumerate}
      \setlength{\itemsep}{6pt}
    \item Wrap a model or function: \texttt{compiled = torch.compile(model)}.
    \item On execution, TorchDynamo captures supported PyTorch operations in graphs with guards on assumptions such as shapes.
    \item For training, PyTorch captures forward and backward regions. The default TorchInductor backend generates optimized code, potentially fusing operations.
    \item Later calls reuse compiled code when the guards hold.
  \end{enumerate}
  \begin{block}{Goal, not guarantee}
    Reduce Python/launch overhead and memory traffic, and choose efficient kernels. Some operations remain library calls; measure the whole step.
  \end{block}
  \smallsource{PyTorch torch.compile documentation}
\end{frame}

\begin{frame}{Compilation: costs, breaks and correctness}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{alertblock}{Why might it not help?}
        First calls include compilation. Graph breaks split regions; changing shapes can trigger recompilation. Some workloads are already dominated by optimized library kernels.
      \end{alertblock}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{How to evaluate}
        Compare eager and compiled with the same model, input and precision. Warm up; synchronize; check loss and gradients within tolerances; inspect the trace.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small \texttt{torch.compile} and efficient SDPA are complementary: SDPA selects an attention implementation; compilation optimizes captured regions around it.
\end{frame}

\begin{frame}{Read one training update on a timeline}
  \small
  \begin{tabular}{@{}p{.20\linewidth}p{.66\linewidth}@{}}
    \toprule
    CPU lane & prepare / launch kernels \hfill next work \\
    GPU lane & copy (if needed) $\to$ forward $\to$ backward $\to$ optimizer \\
    \bottomrule
  \end{tabular}
  \vspace{6pt}
  \begin{itemize}
    \item Zoom from the whole update to a long gap or expensive kernel; link it back to attention, MLP, normalization, loss or optimizer.
    \item Many short kernels suggest a fusion or launch-overhead hypothesis; confirm by comparing traces and timings.
    \item A copy or CPU gap may matter end to end even if model-only device timings look fast.
  \end{itemize}
  \small The lab uses fixed synthetic batches already on the device; it does not time the data loader or host-to-device copies.
\end{frame}

\begin{frame}{Read a profiler trace as a causal story}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Is the GPU idle between steps? Inspect input loading and synchronization.
    \item Are there many tiny kernels? Inspect fusion and compilation opportunities.
    \item Do copies overlap compute? Inspect pinned memory and asynchronous transfer.
    \item Does one kernel dominate? Classify it with shapes, bytes, and FLOPs.
    \item Are unexpected synchronizations present? Find scalar reads and host barriers.
  \end{enumerate}
\end{frame}

\begin{frame}{Apply optimizations in an evidence-driven order}
  \begin{enumerate}
      \setlength{\itemsep}{5pt}
    \item verify correctness and stabilize the benchmark
    \item keep the input pipeline ahead of the GPU
    \item test a supported lower precision with numerical checks
    \item compare manual attention with SDPA
    \item investigate unfused pointwise chains
    \item compare eager with \texttt{torch.compile}
    \item checkpoint activations only when memory is the constraint
  \end{enumerate}
  \begin{alertblock}{One change per measurement}
    Warm up, synchronize, use enough repeated steps, and report median plus dispersion.
  \end{alertblock}
\end{frame}

\begin{frame}{Lab: measure one change in a GPT update}
  \begin{enumerate}
      \setlength{\itemsep}{5pt}
    \item Start with manual FP32 attention; form a bottleneck hypothesis.
    \item Verify the SDPA path exists and compare outputs and gradients before timing it.
    \item Compare supported BF16 or compiled execution against the matching eager baseline.
    \item Report device, shapes, warmup, repeated timings, memory and one limitation.
  \end{enumerate}
  \begin{block}{Scope of the experiment}
    Synthetic batches already on the device isolate model execution. They do not measure the Module 3 loader or end-to-end baseline throughput.
  \end{block}
\end{frame}

\takeaways
{The course GPT update becomes CPU dispatch, GPU kernels and memory traffic.}
{Roofline reasoning separates bandwidth-bound from compute-bound operations.}
{Fusion can reduce intermediate traffic; \texttt{torch.compile} may generate fused kernels.}
{A trustworthy speedup needs stable shapes, warmup, synchronization, correctness checks and profiling evidence.}
