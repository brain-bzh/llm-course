\begin{frame}{What happens when PyTorch runs on CUDA?}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item CPU prepares a batch and issues PyTorch operations.
    \item Tensors are copied to the device when needed; GPU-resident weights and activations stay in device memory.
    \item The CPU enqueues GPU kernels; the GPU executes them asynchronously.
    \item Backward launches more kernels; the optimizer updates weights.
  \end{enumerate}
  \begin{alertblock}{Timing trap}
    A Python timer around an asynchronous launch measures the launches, not the work. Synchronize (wait for the GPU queue to drain) before reading the clock, at the start and at the end of the timed region.
  \end{alertblock}
\end{frame}

\section{Roofline reasoning}

\begin{frame}{Compare implementations: work per byte}
  Both implementations compute $y=(a+b)\times c$ on FP32 arrays.
  \vspace{4pt}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{block}{Two separate kernels}
        Write $a+b$ to memory, then read it back to multiply by $c$.
        \[
          I=\frac{2\ \text{FLOPs}}{24\ \text{bytes}}\approx 0.08
        \]
      \end{block}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{One fused kernel}
        Keep $a+b$ on chip and write only the final result.
        \[
          I=\frac{2\ \text{FLOPs}}{16\ \text{bytes}}=0.125
        \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{3pt}
  \begin{alertblock}{Arithmetic intensity = FLOPs per byte transferred}
    Count bytes at a chosen memory level. Higher intensity helps when bandwidth at that level limits performance.
  \end{alertblock}
\end{frame}

\begin{frame}{Fusion: the same math, fewer intermediates}
  \small
  Consider the pointwise chain $y=\operatorname{ReLU}(x+b)+r$, with equal-sized tensors:
  \vspace{3pt}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{alertblock}{Separate kernels}
        $u=x+b$ \quad $\rightarrow$ write $u$\\[3pt]
        $v=\operatorname{ReLU}(u)$ \quad $\rightarrow$ write $v$\\[3pt]
        $y=v+r$ \quad $\rightarrow$ write $y$
      \end{alertblock}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{One fused kernel}
        Read $x,b,r$; compute the chain; write $y$.\\[5pt]
        $u$ and $v$ stay in registers instead of being written and re-read.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  HBM traffic: 8 versus 4 element transfers.
\end{frame}

\begin{frame}{Why is attention difficult to run fast?}
  \small
  \[
    S=QK^\top/\sqrt{d_h}\quad\longrightarrow\quad
    A=\operatorname{softmax}(S+\text{mask})\quad\longrightarrow\quad O=AV
  \]
  \begin{itemize}
      \setlength{\itemsep}{5pt}
    \item The two matrix products use fast matrix hardware; masking and row-wise softmax add reductions and elementwise work.
    \item Writing and rereading $S$ and $A$, each $T\times T$ per head, costs quadratic storage and HBM traffic.
    \item Irregular tiles, causal masking and limited on-chip memory complicate efficient GPU execution.
  \end{itemize}
  \begin{alertblock}{The target}
    Preserve the attention result while moving fewer intermediates through HBM.
  \end{alertblock}
\end{frame}

\begin{frame}{FlashAttention: tile, fuse, and avoid the large intermediates}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{alertblock}{Straightforward attention}
        $QK^\top$ $\to$ write scores\\[3pt]
        softmax $\to$ write probabilities\\[3pt]
        $AV$ $\to$ write output
      \end{alertblock}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{FlashAttention}
        Load blocks of $Q,K,V$; compute score and value tiles; update online softmax statistics and partial outputs on chip.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small It avoids materializing the full $T\times T$ scores and probabilities in HBM; backward recomputes tiles using saved statistics. Dense attention still has quadratic arithmetic.
  \smallsource{Dao et al., FlashAttention (2022)}
\end{frame}

\begin{frame}{Fused attention behind one call}
  \small
  \begin{enumerate}
      \setlength{\itemsep}{3pt}
    \item Explicit attention: $QK^\top$, mask, softmax and $AV$ are separate kernels with $T\times T$ intermediates.
    \item SDPA computes the same function of $Q,K,V$, each $(B,H,T,d_h)$:\\[1pt]
      {\footnotesize\texttt{y = F.scaled\_dot\_product\_attention(q, k, v, is\_causal=True)}}
    \item The causal mask is a flag, not a tensor. SDPA never disables dropout itself: pass \texttt{dropout\_p=0} in evaluation.
    \item PyTorch picks the backend from hardware, dtype, shapes and mask. FlashAttention needs FP16/BF16 on Ampere or newer (A100); otherwise a fused or math kernel runs.
  \end{enumerate}
  \begin{alertblock}{Same function, different kernels}
    With identical weights, outputs and gradients match the explicit path within tolerance, not bit for bit. Check before timing.
  \end{alertblock}
  \smallsource{PyTorch SDPA documentation; Dao et al., FlashAttention (2022)}
\end{frame}

\checkpoint{roofline diagnosis}
{An optimization halves the FLOP count of a normalization kernel but runtime barely moves. Which roofline quantity explains the result, and what change is more promising?}

\begin{frame}{Roofline diagnosis: one possible answer}
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item The quantity is \textbf{arithmetic intensity}: FLOPs per byte transferred. Normalization is reductions and pointwise work, so its intensity is low and it sits left of the ridge: bandwidth-bound.
    \item There, runtime is roughly bytes moved divided by memory bandwidth. Halving the FLOPs halves the intensity but moves the same bytes.
    \item More promising: move fewer bytes through HBM. Fuse the normalization with neighbouring pointwise operations so intermediates stay on chip, or test a supported lower precision.
    \item Confirm with a trace and repeated timings, not with the FLOP count.
  \end{itemize}
\end{frame}

\begin{frame}{What does \texttt{torch.compile} do?}
  \begin{enumerate}
      \setlength{\itemsep}{6pt}
    \item Wrap a model or function: \texttt{compiled = torch.compile(model)}.
    \item On execution, TorchDynamo captures supported PyTorch operations in graphs with guards on assumptions such as shapes.
    \item For training, PyTorch captures forward and backward regions. The default TorchInductor backend generates optimized code and fuses eligible operations.
    \item Later calls reuse compiled code when the guards hold.
  \end{enumerate}
  \begin{block}{Goal}
    Reduce Python/launch overhead and memory traffic, and choose efficient kernels. Some operations remain library calls.
  \end{block}
  \smallsource{PyTorch torch.compile documentation}
\end{frame}

\begin{frame}{Read one training update on a timeline}
  \small
  \begin{tabular}{@{}p{.20\linewidth}p{.66\linewidth}@{}}
    \toprule
    CPU lane & prepare / launch kernels \hfill next work \\
    GPU lane & copy (if needed) $\to$ forward $\to$ backward $\to$ optimizer \\
    \bottomrule
  \end{tabular}
  \vspace{6pt}
  \begin{itemize}
    \item Zoom from the whole update to a long gap or expensive kernel; link it back to attention, MLP, normalization, loss or optimizer.
    \item Many short kernels suggest launch overhead; fusion may also reduce HBM traffic.
    \item A copy or CPU gap leaves SMs idle, even when individual kernels run fast.
  \end{itemize}
\end{frame}

\begin{frame}{A timing protocol you can defend}
  \small
  \begin{enumerate}
      \setlength{\itemsep}{2pt}
    \item \textbf{One change:} explicit FP32 $\to$ SDPA FP32 $\to$ SDPA BF16; compiled vs eager.
    \item \textbf{Correct first:} outputs, loss and gradients match the baseline within tolerance.
    \item \textbf{Warm up:} exclude the first steps (compilation, allocator growth).
    \item \textbf{Synchronize} before reading the clock, at both ends of the timed region.
    \item \textbf{Repeat:} report the median and the spread, not one sample.
    \item \textbf{Profile separately:} tracing adds overhead; keep it out of timed samples.
    \item \textbf{Report:} device, shapes, tokens/s, peak allocated and reserved memory.
  \end{enumerate}
  \begin{block}{What to report}
    Tokens/s and step time measure throughput; model FLOPs/s divided by the relevant dense peak gives MFU. Device-resident batches isolate model execution.
  \end{block}
\end{frame}





\takeaways
{GPU throughput comes from parallel work, data reuse and enough ready warps.}
{Roofline reasoning relates arithmetic intensity to bandwidth and compute ceilings.}
{Fusion reduces intermediate traffic; \texttt{torch.compile} is one way to get fused kernels.}
{A trustworthy speedup needs stable shapes, warmup, synchronization, correctness checks and profiling evidence.}

\begin{frame}{GEMM: the matrix multiplication behind our model}
  \begin{block}{General matrix--matrix multiplication}
    $Z \leftarrow \alpha XY + \beta Z$, with $X\in\mathbb{R}^{m\times k}$,
    $Y\in\mathbb{R}^{k\times n}$ and $Z\in\mathbb{R}^{m\times n}$.
    A plain matrix product is the case $\alpha=1,\ \beta=0$.
  \end{block}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{block}{In a Transformer}
        Linear layers reshape $B\times T$ token vectors into rows: Q/K/V, attention output, MLP and vocabulary projections.
      \end{block}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{In attention}
        Per-head $QK^\top$ forms scores; $AV$ mixes values. These are batched matrix products around a softmax.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small GEMMs dominate much of the model's arithmetic; shape, dtype and data movement determine speed.
\end{frame}

\begin{frame}{Warps reward regular work}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Good utilization}
        contiguous access\\similar control flow\\enough active warps\\tile-aligned matrix dimensions
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{alertblock}{Lost utilization}
        branch divergence\\strided loads\\register pressure\\tiny or awkward shapes
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \centering Layout, control flow and dimensions affect how much of the hardware does useful work.
\end{frame}



\begin{frame}{Training FLOPs per token for a GPT}
  \small
  Count matrix multiplications only: each weight costs 2 FLOPs per token forward, 4 backward.
  \begin{center}
    \begin{tabular}{@{}lll@{}}
      \toprule
      Term & Matrix weights & FLOPs / token \\
      \midrule
      $Q,K,V$, output projections & $4d^2$ per layer & \\
      MLP $d\to4d\to d$ & $8d^2$ per layer & \\
      Vocabulary head (tied) & $Vd$ & \\
      \midrule
      All projections & $N_{\text{proj}}=12Ld^2+Vd$ & $6N_{\text{proj}}$ \\
      $QK^\top$ and $AV$ products & none & $12\,L\,d\,T$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{-8pt}
  \[
    F_{\text{token}} = 6\,(12Ld^2+Vd)+12\,L\,d\,T
  \]
  \footnotesize The embedding lookup is not a multiplication. Each attention product costs $2Td$ FLOPs per token and layer forward; training triples the forward cost.\par
  \vspace{3pt}
  \textbf{GPT-2 small} ($L=12$, $d=768$, $V=50{,}257$, $T=1024$): $6N_{\text{proj}}\approx7.41\times10^{8}$ plus attention $1.13\times10^{8}$ gives $F_{\text{token}}\approx8.5\times10^{8}$. The $6N$ shortcut gives $7.44\times10^{8}$.
\end{frame}

\begin{frame}{CPU offload: capacity depends on the link}
  \begin{columns}[t]
    \begin{column}{.47\linewidth}
      \begin{block}{GPU memory}
        Active weights and activations\\GPU forward and backward
      \end{block}
    \end{column}
    \begin{column}{.47\linewidth}
      \begin{block}{CPU memory}
        Offloaded state\\possibly a CPU optimizer step
      \end{block}
    \end{column}
  \end{columns}
  \vspace{5pt}
  Tensors cross the host--device link when needed. Transfer time depends on bytes, link bandwidth, and synchronization.
  \begin{alertblock}{Exposed transfers cost step time}
    Transfers that overlap useful compute are hidden; exposed copies or CPU work lengthen the step.
  \end{alertblock}
\end{frame}



\begin{frame}{Vector addition: a bandwidth-bound model}
  For $z=x+y$ with FP32 elements:
  \[
    \text{FLOPs}=1,\qquad \text{bytes}=4+4+4=12
  \]
  \[
    I\approx 0.083\ \text{FLOP/byte}
  \]
  \begin{block}{Prediction}
    A large vector add is bandwidth-bound. A small one is launch-limited instead.
  \end{block}
  \begin{alertblock}{Benchmark check}
    Doubling peak FLOPs should barely change runtime if memory bandwidth stays fixed.
  \end{alertblock}
\end{frame}

\begin{frame}{Large matrix multiplication creates reuse}
  For square $Z=XY$, matrices of width $n$, $b$ bytes per element:
  \[
    \text{FLOPs}\approx2n^3,\qquad \text{minimum bytes}\approx3n^2b
  \]
  \[
    I\approx\frac{2n}{3b}
  \]
  The minimum transfers each matrix element once; tiling supplies the reuse that approaches it.
  \begin{alertblock}{Small matrices behave differently}
    Launch overhead, poor tile occupancy, and insufficient reuse keep small GEMMs far below peak compute.
  \end{alertblock}
\end{frame}

\begin{frame}{Compilation: costs, breaks and correctness}
  \begin{columns}[t]
    \begin{column}{.48\linewidth}
      \begin{alertblock}{When it does not help}
        First calls include compilation. Graph breaks split regions; changing shapes trigger recompilation. Some workloads are already dominated by optimized library kernels.
      \end{alertblock}
    \end{column}
    \begin{column}{.48\linewidth}
      \begin{block}{How to evaluate}
        Compare eager and compiled with the same model, input and precision. Exclude the first calls; check loss and gradients within tolerances.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small \texttt{torch.compile} and efficient SDPA are complementary: SDPA selects an attention implementation; compilation optimizes captured regions around it.
\end{frame}

\begin{frame}{Apply optimizations in an evidence-driven order}
  \begin{enumerate}
      \setlength{\itemsep}{2pt}
    \item verify correctness and stabilize the benchmark
    \item keep the input pipeline ahead of the GPU
    \item test a supported lower precision with numerical checks
    \item compare manual attention with SDPA
    \item investigate unfused pointwise chains
    \item compare eager with \texttt{torch.compile}
    \item checkpoint activations only when memory is the constraint
  \end{enumerate}
  \begin{alertblock}{One change per measurement}
    Warm up, synchronize, use enough repeated steps, and report median plus dispersion. This lecture's unit mappings and traffic counts are models: hardware, dtype, shapes and backend decide the actual kernels, so confirm each gain in the trace.
  \end{alertblock}
\end{frame}

\begin{frame}{Read a profiler trace as a causal story}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Is the GPU idle between steps? Inspect input loading and synchronization.
    \item Are there many tiny kernels? Inspect fusion and compilation opportunities.
    \item Do copies overlap compute? Inspect pinned memory and asynchronous transfer.
    \item Does one kernel dominate? Classify it with shapes, bytes, and FLOPs.
    \item Are unexpected synchronizations present? Find scalar reads and host barriers.
  \end{enumerate}
\end{frame}
