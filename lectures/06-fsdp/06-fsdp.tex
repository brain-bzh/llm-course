\begin{frame}{Sharded AdamW: each rank updates only the slice it owns}
  \small
  View the parameters as one flat vector of $N$ elements, zero-padded to $sW$ with $s=\lceil N/W\rceil$. Rank $r$ owns elements $[\,rs,\ (r+1)s)$ and keeps moments $m,v$ for that slice only.
  \begin{enumerate}
    \item \textbf{Reduce-scatter} the gradients (sum), divide by $W$: mean gradient slice $\bar g$.
    \item \textbf{AdamW on the owned slice} $\theta$, at update count $t$:\par\vspace{-8pt}
      \begin{align*}
        \theta&\leftarrow\theta\,(1-\eta\lambda), \qquad
        m\leftarrow\beta_1 m+(1-\beta_1)\,\bar g, \qquad
        v\leftarrow\beta_2 v+(1-\beta_2)\,\bar g^{2}\\
        \theta&\leftarrow\theta-\frac{\eta}{1-\beta_1^{t}}\;
        \frac{m}{\sqrt{v}\,/\sqrt{1-\beta_2^{t}}+\epsilon}
      \end{align*}
    \item \textbf{All-gather} the updated slices: full parameters on every rank.
  \end{enumerate}
  \vfill
  \textbf{Per-rank optimizer state:} two FP32 moments, $8\lceil N/W\rceil$ bytes against $8N$ in DDP. Padding stays zero: zero gradient, zero moments, zero update.
\end{frame}

\begin{frame}{A memory budget: persistent is not peak}
  \begin{block}{Example: 1B parameters, four ranks, 16 bytes / parameter}
    DDP: $16$ GB per rank \quad $\longrightarrow$ \quad ideal full sharding: $16/4=4$ GB per rank
  \end{block}
  \small Decimal GB.
  \vfill
  \begin{alertblock}{What else occupies the peak?}
    Temporary full units, including a next unit gathered early (prefetch), activations, collective buffers, and allocator overhead. The local shard is already counted: add only the extra materialized storage.
  \end{alertblock}
\end{frame}

\checkpoint{wrapping policy}
{A run fits in memory but launches hundreds of tiny all-gathers and scales poorly. What wrapping change would you test, and which memory and timing quantities could regress?}

\begin{frame}{Wrapping policy: one possible answer}
  \begin{itemize}
      \setlength{\itemsep}{9pt}
    \item Test coarser units: wrap each Transformer block (or a group of blocks) instead of small submodules. Fewer, larger all-gathers are less latency-dominated.
    \item Memory that could regress: peak transient materialization, since a larger full unit is gathered (plus the next one when prefetching), and memory is released more coarsely.
    \item Timing that could regress: fewer opportunities for overlap, so a longer exposed collective.
    \item Compare peak allocated memory, collective exposure and tokens per second, with everything else fixed.
  \end{itemize}
\end{frame}

\section{Choosing and comparing approaches}

\begin{frame}{A DDP--FSDP comparison needs a fixed contract}
  Match model, global token batch, optimizer, schedule, precision, data order, checkpoint cadence, and evaluation.
  \vspace{7pt}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Correctness evidence}
        first-update comparison\\loss trajectory\\resume equivalence
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{Systems evidence}
        peak allocated memory\\tokens per second\\collective exposure\\checkpoint time and size
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Choose FSDP when memory is the binding constraint}
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item Prefer DDP when the model fits comfortably and communication simplicity wins.
    \item Prefer FSDP when replicated state blocks model size, batch size, or context.
    \item Tune wrapping and prefetch from traces, not from a universal recipe.
    \item Treat checkpoint load and resharding as part of the production design.
  \end{itemize}
\end{frame}

\begin{frame}{A distributed checkpoint is a data layout}
  \begin{columns}[t]
    \begin{column}{0.31\linewidth}
      \begin{block}{Full}
        one file, gathered on rank 0\\loads anywhere\\rank 0 memory pressure
      \end{block}
    \end{column}
    \begin{column}{0.31\linewidth}
      \begin{block}{Sharded}
        each rank writes its slices in parallel\\tied to the saving world size
      \end{block}
    \end{column}
    \begin{column}{0.31\linewidth}
      \begin{block}{Sharded + metadata}
        records each slice's offset in the global tensor\\reloads at another world size
      \end{block}
    \end{column}
  \end{columns}
  \begin{alertblock}{Completion protocol}
    Publish a checkpoint only after every shard and its metadata are durable. A directory that exists is not proof of a complete save.
  \end{alertblock}
\end{frame}

\checkpoint{memory prediction}
{Estimate idealized model-state memory per rank for $N$ parameters, 16 bytes per parameter, and world size $W$. Then name three reasons the measured peak will be higher.}

\begin{frame}{Memory prediction: one possible answer}
  \begin{itemize}
      \setlength{\itemsep}{9pt}
    \item Unsharded model state: $16N$ bytes. Ideal full sharding: $16N/W$ bytes per rank.
    \item Check with the earlier example: $N=10^9$, $W=4$ gives $16/4=4$ GB per rank.
    \item Higher in practice, reason 1: full units gathered temporarily, plus any prefetched unit.
    \item Reason 2: activations, which sharding does not reduce.
    \item Reason 3: collective buffers and allocator overhead.
  \end{itemize}
\end{frame}

\begin{frame}{Keep or reshard after forward?}
  \begin{columns}[t]
    \begin{column}{.47\linewidth}
      \begin{block}{Keep full parameters}
        Gather $\to$ forward $\to$ backward\\[5pt]

      \end{block}
    \end{column}
    \begin{column}{.47\linewidth}
      \begin{alertblock}{Reshard after forward}
        Gather $\to$ forward $\to$ reshard $\to$ gather $\to$ backward\\[5pt]
        Less residency between uses; another gather.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  \small The explorer uses the first policy; the lifecycle figure (``Sharding changes when full parameters exist'') shows the second.
\end{frame}

\begin{frame}{Two collectives make sharding usable}
  \centering\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    & Rank 0 & Rank 1 \\
    \midrule
    Start: parameter shards & $P_0$ & $P_1$ \\
    \textbf{All-gather} parameters & $[P_0,P_1]$ & $[P_0,P_1]$ \\
    Local forward / backward & gradient $[g_{00},g_{01}]$ & gradient $[g_{10},g_{11}]$ \\
    \textbf{Reduce-scatter} gradients & $g_{00}+g_{10}$ & $g_{01}+g_{11}$ \\
    \bottomrule
  \end{tabular}
  \vfill
  \begin{block}{The division of work}
    Each rank gathers the full unit to compute its own data; after reduction it retains the gradient for its owned shard. Averaging requires the appropriate world-size normalization.
  \end{block}
\end{frame}
\begin{frame}{CPU offload and sharding can work together}
  \begin{block}{A second memory lever}
    Sharding distributes model state across ranks. Offloading places selected state in CPU memory; the two can be combined.
  \end{block}
  \begin{itemize}
    \item Example: offload optimizer state and run the optimizer on CPU.
    \item Other policies offload parameters and gradients too.
    \item GPU--CPU traffic, CPU work and synchronization reduce throughput.
  \end{itemize}
  \small Offload is a capacity tool, not a free speedup.
\end{frame}
