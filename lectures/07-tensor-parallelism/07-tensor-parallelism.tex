\begin{frame}{Why column then row?}
  \begin{block}{The layouts match across the activation}
    Column parallelism produces feature shards; the activation and row-parallel layer use them locally.
  \end{block}
  \vspace{5pt}
  \begin{alertblock}{Why not row first?}
    Partial values must be summed before the nonlinearity: $\phi(Z_0+Z_1)\ne\phi(Z_0)+\phi(Z_1)$ in general.
  \end{alertblock}
  \vfill
  \small With replicated input and output, column $\to$ row keeps the intermediate sharded without an extra collective. Other pairings require different communication or assumptions.
\end{frame}

\begin{frame}{Forward and backward collectives are conjugate}
  \small
  \centering
  \begin{tabular}{@{}lll@{}}
    \toprule
    Forward operation & Layout change & Backward \\
    \midrule
    identity on replicated $X$ & replicated $\to$ sharded use & all-reduce $\partial\mathcal{L}/\partial X$ \\
    all-reduce sum & partial $\to$ replicated & identity \\
    reduce-scatter & partials $\to$ token shards & all-gather \\
    all-gather & token shards $\to$ replicated & reduce-scatter \\
    \bottomrule
  \end{tabular}
  \par\vfill\vspace{4pt}\raggedright
  \textbf{Column input:} $\partial\mathcal{L}/\partial X=\sum_{r}(\partial\mathcal{L}/\partial Y_r)\,W_r^{\top}$; rank $r$ holds only its own term, so backward sums.\par\smallskip
  \textbf{Row output:} every rank already holds the same $\partial\mathcal{L}/\partial Y$; a second sum would multiply it by $N_{tp}$.
  \vfill
  \begin{alertblock}{Testing rule}
    Compare outputs, input and weight gradients with an unsharded reference. A missing backward all-reduce breaks only the input gradient.
  \end{alertblock}
\end{frame}

\checkpoint{layout trace}
{Trace replicated or sharded status through a column-parallel MLP expansion, activation, row-parallel contraction, and residual add. At which point is communication unavoidable?}

\begin{frame}{Layout trace: one possible answer}
  \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item MLP input: replicated, $B\times T\times d$.
    \item Column-parallel expansion: feature sharded, $B\times T\times 4d/N_{tp}$; no forward collective.
    \item Activation: elementwise, so it runs locally and the layout stays sharded.
    \item Row-parallel contraction: each rank holds a \emph{partial value} of full shape $B\times T\times d$.
    \item \textbf{Unavoidable:} the all-reduce that sums the partials before the residual add, which needs the complete, replicated value. Backward mirrors it with an all-reduce of the input gradient.
  \end{itemize}
\end{frame}

\section{Cost and placement}

\begin{frame}{Communication cost depends on activation size}
  A collective over $B\times T\times d$ elements moves bytes proportional to the activation tensor, not the parameter matrix.
  \[
    S_{\text{act}}=B\,T\,d\,b
    \qquad (b\text{: bytes per element})
  \]
  \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item Larger batch and context increase collective payload.
    \item Higher TP degree shrinks local matmuls and can reduce arithmetic efficiency.
    \item Collective latency and link bandwidth set the exposed cost.
  \end{itemize}
\end{frame}

\begin{frame}{Tensor parallelism belongs on the fastest links}
  \begin{columns}[c]
    \begin{column}{0.48\linewidth}
      \begin{block}{Intra-node}
        High-bandwidth, low-latency GPU fabrics suit frequent activation collectives.
      \end{block}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{alertblock}{Across slow links}
        Small per-rank matmuls plus exposed all-reduces can erase the benefit quickly.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vfill
  Keep TP inside a fast island when possible, then use data or pipeline parallelism across nodes.
\end{frame}

\begin{frame}{Choose the smallest degree that solves the constraint}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Determine whether a layer's weights or activations fail to fit on one GPU.
    \item Check head, hidden, and MLP divisibility for candidate degrees.
    \item Map the group to the strongest available links.
    \item Measure local matmul efficiency and collective exposure.
    \item Verify numerics against the unsharded block.
  \end{enumerate}
\end{frame}

\checkpoint{degree selection}
{A model fits with TP=2, but TP=8 gives lower throughput. Explain the result using local matrix shapes, collective frequency, and topology.}

\begin{frame}{Degree selection: one possible answer}
  \small
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item \textbf{Local matrix shapes:} each rank's share of the features falls from $1/2$ to $1/8$ (MLP expansion width $4d/2\to4d/8$). Smaller matmuls use the hardware less efficiently.
    \item \textbf{Collective frequency:} every block still needs its activation all-reduces, each over $B\times T\times d$ elements whatever the degree. Local compute shrinks but communication does not, so more of it is exposed.
    \item \textbf{Topology:} a two-rank group fits on the fastest links; an eight-rank group involves more ranks per collective and may reach slower links.
    \item \textbf{Conclusion:} the model already fits at TP=2, so keep the smallest degree that solves the constraint and give the remaining GPUs to data parallelism.
  \end{itemize}
\end{frame}

\takeaways
{Column parallelism shards output features, while row parallelism sums partial outputs.}
{Their composition keeps intermediate MLP and attention shards local.}
{A layout ledger and unsharded numerical reference are essential correctness tools.}
{Tensor parallelism is a fine-grained method that needs large local work and fast links.}
