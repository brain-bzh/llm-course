\begin{frame}{Two techniques shard the sequence axis for different reasons}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Sequence parallelism}
        Complements tensor parallelism by sharding token activations around norms, dropout, and elementwise operations.
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{Context parallelism}
        Splits a long context across devices so each rank stores only part of the queries and token activations.
      \end{block}
    \end{column}
  \end{columns}
  \begin{alertblock}{The hard part}
    Attention couples every local query with keys and values that may live on every other rank.
  \end{alertblock}
\end{frame}

\begin{frame}{Most layers are local after token sharding}
  With $T$ split over $C$ context ranks, each rank owns roughly $T/C$ positions.
  \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item tokenwise normalization and MLPs operate locally
    \item residual additions operate locally
    \item attention needs remote key-value blocks
    \item some losses and metrics need global reductions
  \end{itemize}
  \begin{block}{Memory effect}
    Token activation memory falls roughly with $1/C$, while communication appears inside every attention layer.
  \end{block}
\end{frame}

\begin{frame}{Online softmax makes blockwise attention exact}
  For one query row, maintain running maximum $m$, normalizer $\ell$, and numerator $o$.
  For a new score block with maximum $m_b$:
  \[
    m'=\max(m,m_b)
  \]
  \[
    \ell'=e^{m-m'}\ell+\sum_j e^{s_j-m'}
  \]
  \[
    o'=e^{m-m'}o+\sum_j e^{s_j-m'}v_j
  \]
  Final attention is $o/\ell$. Rescaling preserves contributions computed under earlier maxima.
\end{frame}

\begin{frame}{CP communication: stream KV, then synchronize gradients}
  \small Let $C$ ranks split $T$ tokens; each local KV block contains $B(T/C)d_{kv}$ elements per K or V.
  \begin{block}{Illustrative forward ring payload per rank}
    \[
      (C-1)\times 2B(T/C)d_{kv}\,b\quad\text{bytes sent}
    \]
    $b$ is bytes/element; $d_{kv}$ is total KV width. Assumes $C-1$ full-block sends and equal chunks.
  \end{block}
  \begin{itemize}
    \item Causal skipping, GQA, block layout and implementation change actual traffic and compute.
    \item Backward exchanges information for gradients; replicated weights require gradient synchronization across CP ranks.
    \item Overlap can hide some transfer, but exposed cost depends on link bandwidth and latency.
  \end{itemize}
\end{frame}

\section{Pipeline parallelism}

\begin{frame}{Pipeline parallelism assigns layer ranges to stages}
  \begin{center}
    \begin{tikzpicture}[node distance=.55cm,every node/.style={font=\small},box/.style={draw=themebluedraw,fill=themebluebg,rounded corners,align=center,minimum width=2.2cm,minimum height=.7cm}]

      \node[box,right=of s1] (s2) {stage 2\\ layers 12--17};
      \node[box,right=of s2] (s3) {stage 3\\ layers 18--23};
      \draw[-{Stealth[length=4pt,width=3pt]},thick,draw=themenavy] (s0)--(s1);
      \draw[-{Stealth[length=4pt,width=3pt]},thick,draw=themenavy] (s1)--(s2);
      \draw[-{Stealth[length=4pt,width=3pt]},thick,draw=themenavy] (s2)--(s3);
    \end{tikzpicture}
  \end{center}
  \begin{block}{Why micro-batches?}
    Split a global batch into $M$ micro-batches so different stages can work on different micro-batches concurrently.
  \end{block}
\end{frame}

\begin{frame}{Pipeline bubbles shrink as micro-batches increase}
  For $P$ balanced stages, $M$ micro-batches, and negligible communication:
  \[
    \text{ideal utilization}\approx\frac{M}{M+P-1}
  \]
  \begin{block}{Worked example}
    $P=4$, $M=16$ gives idle fraction $3/19\approx15.8\%$.\\
    Below $10\%$ requires $M>9(P-1)$, hence at least 28 micro-batches here.
  \end{block}
  \begin{alertblock}{Trade-off}
    More micro-batches reduce bubbles but can create smaller, less efficient kernels and more activation bookkeeping.
  \end{alertblock}
\end{frame}

\begin{frame}{AFAB and 1F1B trade memory for schedule complexity}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{All-forward-all-backward}
        simple schedule\\many outstanding activations\\higher peak memory
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{One-forward-one-backward}
        after warmup, alternate directions\\fewer stored activations\\more coordination
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  Both still have warmup and cooldown bubbles. Interleaving virtual stages can reduce them further at added complexity.
\end{frame}

\begin{frame}{The slowest stage sets pipeline throughput}
  \[
    t_{\text{cycle}}\approx\max_s t_s
  \]
  \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item Equal layer counts need not mean equal work.
    \item Embeddings, output projection, and activation sizes can skew stages.
    \item Stage boundaries determine activation communication volume.
    \item Profile stage time and memory before moving boundaries.
  \end{itemize}
\end{frame}

\section{Expert parallelism}

\begin{frame}{From one dense MLP to routed experts}
  \begin{columns}[c]
    \begin{column}{.43\linewidth}
      \begin{block}{Dense MLP}
        Every token passes through the same MLP.
      \end{block}
    \end{column}
    \begin{column}{.53\linewidth}
      \begin{block}{Mixture of experts}
        A router scores $E$ expert MLPs and selects top-$k$ for each token. Combine the selected outputs using routing weights.
      \end{block}
    \end{column}
  \end{columns}
  \[
    y(x)=\sum_{e\in\operatorname{top-}k(x)}p_e(x)\,\operatorname{MLP}_e(x)
  \]
  \small Total expert parameters grow with $E$; active expert work per token depends on $k$ and expert size. EP distributes expert weights across GPUs.
\end{frame}

    \node[local] (out) at (9.8,0) {weighted sum\\restore order};

  \end{tikzpicture}
  \end{center}
  \begin{block}{Why all-to-all?}
    Every source rank can send different token groups to every expert rank. Backward reverses the data movement and propagates gradients through selected experts and routing weights.
  \end{block}
  \small Group size, skew, message size and topology determine cost; implementations may use other dispatch primitives or fuse steps.
\end{frame}

\begin{frame}{EP communication grows with active routes}
  \small Let each rank originate $n$ tokens of width $d$, routed to $k$ experts; each element uses $b$ bytes.
  \begin{block}{Simple forward payload per source rank}
    \[
      \text{dispatch + return}\ \approx\ 2n k d b\quad\text{bytes}
    \]
    Counts one full vector per assignment in each direction. Uniform routing over $E_P$ ranks gives a remote fraction near $1-1/E_P$.
  \end{block}
  \begin{itemize}
    \item Backward adds traffic; metadata, padding, skew and protocol overhead are excluded.
    \item Local routes do not traverse the network; coalescing and TP/SP layouts can alter payloads.
    \item Measure exposed dispatch/combine time and expert GEMM efficiency, not only average bytes.
  \end{itemize}
\end{frame}

\begin{frame}{Load balance is an MoE performance constraint}
  \begin{columns}[t]
    \begin{column}{.47\linewidth}
      \begin{alertblock}{Skewed routing}
        One expert receives many tokens: its GPU becomes the straggler, buffers grow, and small experts run inefficient GEMMs.
      \end{alertblock}
    \end{column}
    \begin{column}{.47\linewidth}
      \begin{block}{Ways to respond}
        Auxiliary balancing losses or router biases; capacity limits and optional padding/dropping; better dispatch and placement.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small Monitor tokens per expert and rank, maximum / mean load, dropped tokens if applicable, and all-to-all time. Balancing can affect model behavior, not just throughput.
\end{frame}

\section{Composing parallel dimensions}

\begin{frame}{The device count factorizes across parallel groups}
  \[
    N_{\text{GPU}}=D\times T_p\times C\times P
  \]
  \small
  \begin{tabular}{lll}
    \toprule
    Dimension & Splits & Communication pattern \\
    \midrule
    data & independent micro-batches & gradient or shard collectives \\
    tensor & hidden width and heads & frequent activation collectives \\
    context & sequence positions & key-value exchange per layer \\
    pipeline & layer depth & point-to-point activations \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{Subgroups reuse existing devices}
  \begin{itemize}
      \setlength{\itemsep}{9pt}
    \item $D$ counts the total data-parallel degree.
    \item FSDP degree $F$ divides $D$: $D/F$ groups replicate state shards.
    \item EP degree $E_P$ can divide $D$: each expert has $D/E_P$ replicas.
    \item Shared and expert parameters use different synchronization groups.
  \end{itemize}
  \begin{block}{Eight-device example}
    $D=4$, $T_p=2$, $C=P=1$. Setting $E_P=4$ still uses eight devices.\\
    Global tokens/update: $B_\mu\,T\,A\,D$.
  \end{block}
  \smallsource{Teaching layout: combined EP/FSDP requires separate expert shard-group definitions}
\end{frame}

\begin{frame}{Map the chattiest group to the fastest fabric}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Start by placing frequent TP collectives on a fast fabric.
    \item Add context parallelism when token activations or attention exceed local memory.
    \item Use pipeline parallelism when model depth spans nodes.
    \item Replicate the resulting model group with data parallelism when batch and devices remain.
  \end{enumerate}
  \begin{alertblock}{Selection rule}
    State the rank layout and measure exposed transfer time. Persistent state excludes activations, gathered weights, and communication buffers.
  \end{alertblock}
\end{frame}

\begin{frame}{Four cases lead to different strategies}
  \small
  \begin{tabular}{p{.41\linewidth}p{.49\linewidth}}
    \toprule
    Binding constraint & First strategy to evaluate \\
    \midrule
    model fits, short context, more GPUs & DDP or FSDP replicas \\
    one wide layer does not fit & tensor parallelism \\
    weights fit, 128k-token examples do not & context parallelism plus checkpointing \\
    model exceeds one node & pipeline across nodes, TP within node \\
    \bottomrule
  \end{tabular}
\end{frame}

\takeaways
