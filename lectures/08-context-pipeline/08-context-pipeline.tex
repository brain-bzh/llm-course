\begin{frame}{Two techniques shard the sequence axis for different reasons}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Sequence parallelism}
        Complements tensor parallelism by sharding token activations around norms, dropout, and elementwise operations.
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{Context parallelism}
        Splits a long context across devices so each rank stores only part of the queries and token activations.
      \end{block}
    \end{column}
  \end{columns}
  \begin{alertblock}{The hard part}
    Attention couples every local query with keys and values that may live on every other rank.
  \end{alertblock}
\end{frame}

\begin{frame}{Most layers are local after token sharding}
  With $T$ split over $N_{cp}$ context ranks, each rank owns roughly $T/N_{cp}$ positions.
  \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item tokenwise normalization and MLPs operate locally
    \item residual additions operate locally
    \item attention needs remote key-value blocks
    \item some losses and metrics need global reductions
  \end{itemize}
  \begin{block}{Memory effect}
    Token activation memory falls roughly with $1/N_{cp}$, while communication appears inside every attention layer.
  \end{block}
\end{frame}

\begin{frame}{Online softmax makes blockwise attention exact}
  For one query row, maintain running maximum $m$, normalizer $\ell$, and numerator $o$.
  For a new score block with maximum $m_b$:
  \[
    m'=\max(m,m_b)
  \]
  \[
    \ell'=e^{m-m'}\ell+\sum_j e^{s_j-m'}
  \]
  \[
    o'=e^{m-m'}o+\sum_j e^{s_j-m'}v_j
  \]
  Final attention is $o/\ell$. Rescaling preserves contributions computed under earlier maxima.
\end{frame}

    \end{tikzpicture}

  \begin{block}{Why micro-batches?}
    Split a global batch into $M$ micro-batches so different stages can work on different micro-batches concurrently.
  \end{block}
\end{frame}

\begin{frame}{Pipeline bubbles shrink as micro-batches increase}
  For $N_{pp}$ balanced stages, $M$ micro-batches, and negligible communication, with AFAB or 1F1B:
  \[
    \text{utilization}=\frac{M}{M+N_{pp}-1},
    \qquad
    \text{bubble (idle) fraction}=\frac{N_{pp}-1}{M+N_{pp}-1}
  \]
  \begin{block}{Worked example}
    $N_{pp}=4$, $M=16$ gives idle fraction $3/19\approx15.8\%$.\\
    Below $10\%$ requires $M>9(N_{pp}-1)$, hence at least 28 micro-batches here.
  \end{block}
  \begin{alertblock}{Trade-off}
    More micro-batches reduce bubbles but create smaller, less efficient kernels and more activation bookkeeping.
  \end{alertblock}
\end{frame}

\checkpoint{schedule arithmetic}
{For six stages and twelve micro-batches, estimate idealized forward utilization. Name two real effects that make measured utilization lower.}

\begin{frame}{Schedule arithmetic: one possible answer}
  With $N_{pp}=6$ stages and $M=12$ micro-batches:
  \[
    \frac{M}{M+N_{pp}-1}=\frac{12}{12+6-1}=\frac{12}{17}\approx70.6\%
  \]
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item The idle fraction is $5/17\approx29.4\%$.
    \item Real effect 1: boundary transfers of activations and gradients are not free.
    \item Real effect 2: stages are rarely balanced, and the slowest stage sets the cycle time.
    \item Also acceptable: smaller micro-batches run less efficient kernels.
  \end{itemize}
\end{frame}

\section{Expert parallelism and composition}

\begin{frame}{From one dense MLP to routed experts}
  \begin{columns}[c]
    \begin{column}{.43\linewidth}
      \begin{block}{Dense MLP}
        Every token passes through the same MLP.
      \end{block}
    \end{column}
    \begin{column}{.53\linewidth}
      \begin{block}{Mixture of experts}
        A router scores $E$ expert MLPs and selects top-$k$ for each token. Combine the selected outputs using routing weights.
      \end{block}
    \end{column}
  \end{columns}
  \[
    y(x)=\sum_{e\in\operatorname{top-}k(x)}p_e(x)\,\operatorname{MLP}_e(x)
  \]
  \small Total expert parameters grow with $E$; active expert work per token depends on $k$ and expert size. EP distributes expert weights across GPUs: tokens are dispatched to the GPUs owning their experts and outputs return (all-to-all).
\end{frame}

\begin{frame}{The device count factorizes across parallel groups}
  \[
    W=N_{dp}\times N_{tp}\times N_{cp}\times N_{pp}
  \]
  \par\vspace{-4pt}
  {\centering\footnotesize $W$: world size (devices); $N_{dp},N_{tp},N_{cp},N_{pp}$: data, tensor, context and pipeline degrees\par}
  \small
  \begin{tabular}{lll}
    \toprule
    Dimension & Splits & Communication pattern \\
    \midrule
    data & independent micro-batches & gradient or shard collectives \\
    tensor & hidden width and heads & frequent activation collectives \\
    context & sequence positions & key-value exchange per layer \\
    pipeline & layer depth & point-to-point activations \\
    \bottomrule
  \end{tabular}
  \vfill
  Each rank has one coordinate per dimension. A group for one dimension is the set of ranks that differ only in that coordinate.
\end{frame}

\begin{frame}{Subgroups reuse existing devices}
  \begin{itemize}
      \setlength{\itemsep}{9pt}
    \item $N_{dp}$ counts the total data-parallel degree.
    \item FSDP degree $N_{fsdp}$ divides $N_{dp}$: $N_{dp}/N_{fsdp}$ groups replicate state shards.
    \item EP degree $N_{ep}$ can divide $N_{dp}$: each expert has $N_{dp}/N_{ep}$ replicas.
    \item Shared and expert parameters use different synchronization groups.
  \end{itemize}
  \begin{block}{Eight-device example}
    $N_{dp}=4$, $N_{tp}=2$, $N_{cp}=N_{pp}=1$. Setting $N_{ep}=4$ or $N_{fsdp}=4$ still uses eight devices.
  \end{block}
  \smallsource{Teaching layout: combined EP/FSDP requires separate expert shard-group definitions}
\end{frame}

\begin{frame}{Persistent state per rank follows the layout}
  \small
  \begin{center}
    \begin{tabular}{@{}lll@{}}
      \toprule
      Degree & Devices & Parameters stored per rank \\
      \midrule
      data $N_{dp}$ & $\times N_{dp}$ & unchanged: full replicas \\
      context $N_{cp}$ & $\times N_{cp}$ & unchanged: splits activations only \\
      tensor $N_{tp}$ & $\times N_{tp}$ & divided by $N_{tp}$ \\
      pipeline $N_{pp}$ & $\times N_{pp}$ & divided by $N_{pp}$ \\
      FSDP $N_{fsdp}$ & none: divides $N_{dp}$ & divided by $N_{fsdp}$ \\
      expert $N_{ep}$ & none: divides $N_{dp}$ & expert parameters divided by $N_{ep}$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \[
    N_{\text{rank}}=\frac{N_{\text{shared}}+N_{\text{expert}}/N_{ep}}{N_{tp}\,N_{pp}\,N_{fsdp}},
    \qquad
    \text{tokens per update}=B\,T\,M\,N_{dp}
  \]
  \vfill
  Persistent bytes are $N_{\text{rank}}$ times the bytes per parameter for weights, gradients and optimizer state. Only data parallelism multiplies the batch: $M$ micro-batches of $B$ sequences per replica.
\end{frame}

\begin{frame}{Map the chattiest group to the fastest fabric}
  \begin{enumerate}
      \setlength{\itemsep}{7pt}
    \item Start by placing frequent TP collectives on a fast fabric.
    \item Add context parallelism when token activations or attention exceed local memory.
    \item Use pipeline parallelism when model depth spans nodes.
    \item Replicate the resulting model group with data parallelism when batch and devices remain.
  \end{enumerate}
  \begin{alertblock}{Selection rule}
    State the rank layout and measure exposed transfer time. Persistent state excludes activations, gathered weights, and communication buffers.
  \end{alertblock}
\end{frame}

\begin{frame}{Four cases lead to different strategies}
  \small
  \begin{tabular}{p{.41\linewidth}p{.49\linewidth}}
    \toprule
    Binding constraint & First strategy to evaluate \\
    \midrule
    model fits, short context, more GPUs & DDP or FSDP replicas \\
    one wide layer does not fit & tensor parallelism \\
    weights fit, 128k-token examples do not & context parallelism plus checkpointing \\
    model exceeds one node & pipeline across nodes, TP within node \\
    \bottomrule
  \end{tabular}
\end{frame}

\checkpoint{cluster design}
{Design a layout for 32 GPUs arranged as four 8-GPU nodes. State the memory constraint, proposed $N_{dp},N_{tp},N_{cp},N_{pp}$, and which communication crosses node boundaries.}

\begin{frame}{Cluster design: one possible answer}
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item \textbf{Assumed constraint:} the model's state and activations exceed one 8-GPU node but fit on two; contexts are short.
    \item \textbf{Layout:} $N_{tp}=8$, $N_{pp}=2$, $N_{cp}=1$, $N_{dp}=2$, so $W=N_{dp}\times N_{tp}\times N_{cp}\times N_{pp}=2\times8\times1\times2=32$.
    \item \textbf{Inside a node:} the frequent tensor-parallel activation collectives.
    \item \textbf{Across nodes:} point-to-point pipeline activations and gradients between the two stages of a replica, and gradient synchronization between the two replicas.
  \end{itemize}
  \vfill
  \small A different constraint gives a different layout: a model that fits on one GPU would use $N_{dp}=32$.
\end{frame}

\takeaways
{Context parallelism shards tokens and uses exact blockwise attention with online softmax.}
{Pipeline parallelism shards depth, while micro-batches trade bubble size against memory and kernel efficiency.}
{Expert parallelism routes tokens through distributed MLPs using dispatch and return communication.}
{Count independent axes once and test placement against measured memory and communication.}

\begin{frame}{CP communication: stream KV, then synchronize gradients}
  \small Let $N_{cp}$ ranks split $T$ tokens; each local KV block contains $B(T/N_{cp})H_{kv}d_h$ elements per K or V.
  \begin{block}{Forward ring payload per rank}
    \[
      (N_{cp}-1)\times 2B(T/N_{cp})H_{kv}d_h\,b\quad\text{bytes sent}
    \]
    $b$ is bytes/element; $H_{kv}d_h$ is total KV width. Assumes $N_{cp}-1$ full-block sends and equal chunks.
  \end{block}
  \begin{itemize}
    \item Causal skipping, GQA, block layout and implementation change actual traffic and compute.
    \item Backward exchanges information for gradients; replicated weights require gradient synchronization across CP ranks.
    \item Overlap hides some transfer; link bandwidth and latency set the exposed cost.
  \end{itemize}
\end{frame}

\begin{frame}{The slowest stage sets pipeline throughput}
  \[
    t_{\text{cycle}}\approx\max_s t_s
  \]
  \begin{itemize}
      \setlength{\itemsep}{7pt}
    \item Equal layer counts need not mean equal work.
    \item Embeddings, output projection, and activation sizes can skew stages.
    \item Stage boundaries determine activation communication volume.
    \item Move boundaries to balance stage time and memory.
  \end{itemize}
\end{frame}

    \node[local] (expert) at (4.9,0) {local\\expert MLPs};

    \node[local] (out) at (9.8,0) {weighted sum\\restore order};
    \draw[arrow] (route)--(dispatch); \draw[arrow] (dispatch)--(expert); \draw[arrow] (expert)--(combine); \draw[arrow] (combine)--(out);
  \end{tikzpicture}
  \end{center}
  \begin{block}{Why all-to-all?}
    Every source rank can send different token groups to every expert rank. Backward reverses the data movement and propagates gradients through selected experts and routing weights.
  \end{block}
  \small Group size, skew, message size and topology determine cost.
\end{frame}

\begin{frame}{EP communication grows with active routes}
  \small Let each rank originate $n$ tokens of width $d$, routed to $k$ experts; each element uses $b$ bytes.
  \begin{block}{Simple forward payload per source rank}
    \[
      \text{dispatch + return}\ \approx\ 2n k d b\quad\text{bytes}
    \]
    Counts one full vector per assignment in each direction. Uniform routing over $N_{ep}$ ranks gives a remote fraction near $1-1/N_{ep}$.
  \end{block}
  \begin{itemize}
    \item Backward adds traffic; metadata, padding, skew and protocol overhead are excluded.
    \item Local routes do not traverse the network.
    \item Step time follows exposed dispatch/combine time and expert GEMM efficiency, not only average bytes.
  \end{itemize}
\end{frame}

\begin{frame}{Load balance is an MoE performance constraint}
  \begin{columns}[t]
    \begin{column}{.47\linewidth}
      \begin{alertblock}{Skewed routing}
        One expert receives many tokens: its GPU becomes the straggler, buffers grow, and small experts run inefficient GEMMs.
      \end{alertblock}
    \end{column}
    \begin{column}{.47\linewidth}
      \begin{block}{Ways to respond}
        Auxiliary balancing losses or router biases; capacity limits and optional padding/dropping; better dispatch and placement.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small Monitor tokens per expert and rank, maximum / mean load, dropped tokens if applicable, and all-to-all time. Balancing can affect model behavior, not just throughput.
\end{frame}
