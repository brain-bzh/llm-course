\section{Why caching works: prefill and decode}

\begin{frame}{Naive decoding recomputes the unchanged prefix}
  \centering

    \node[anchor=east,font=\small\bfseries] at (-.85,.5) {step 3};
    \foreach \x/\v in {0/$x_1$,1/$x_2$,2/$x_3$}{\node[tok] at (\x,2.7) {\v}; \node[tok] at (\x,1.6) {\v}; \node[tok] at (\x,.5) {\v};}
    \node[new] at (3,1.6) {$y_1$}; \node[new] at (3,.5) {$y_1$}; \node[new] at (4,.5) {$y_2$};
    \draw[-{Stealth[length=4pt]},thick] (2.5,2.7)--(5.4,2.7) node[right]{sample $y_1$};
    \draw[-{Stealth[length=4pt]},thick] (3.5,1.6)--(5.4,1.6) node[right]{sample $y_2$};
    \draw[-{Stealth[length=4pt]},thick] (4.5,.5)--(5.4,.5) node[right]{sample $y_3$};
  \end{tikzpicture}
  \par\vspace{5pt}
  \begin{block}{Why the prefix never changes}
    With a causal mask, position $j$ attends only to positions $\le j$. With fixed weights and causal attention, earlier states and their keys and values are unchanged mathematically. Different floating-point execution paths can differ slightly.
  \end{block}
  \footnotesize Example: three prompt tokens and two generated tokens, with fixed weights.
\end{frame}

    \node[attn] (attention) at (8.55,2.35) {attention\\$q_t$ with $K,V$};
    \node[proj] (out) at (10.9,2.35) {output\\$o_t$};

  \[
    o_t=\operatorname{softmax}\!\left(q_tK_{1:t}^{\top}/\sqrt{d_h}\right)V_{1:t}

\end{frame}

\begin{frame}{One request: one prefill, then one token per step}
  \centering
  \begin{tikzpicture}[scale=.9,transform shape,every node/.style={font=\scriptsize,align=center},
    input/.style={draw=themebluedraw,fill=themebluebg,minimum width=1.7cm,minimum height=.8cm},
    compute/.style={draw=accentdarkdraw,fill=accentdarkbg,minimum width=1.8cm,minimum height=.8cm},
    token/.style={draw=leafreddraw,fill=leafredbg,minimum width=1.0cm,minimum height=.8cm},
    arr/.style={-{Stealth[length=4pt]},thick}]
    \node[input] (p) at (0,1.9) {prompt\\$x_{1:P}$};
    \node[compute] (prefill) at (2.5,1.9) {prefill\\$P$ queries};
    \node[token] (y1) at (4.9,1.9) {sample\\$y_1$};
    \node[compute] (decode) at (7.1,1.9) {decode\\1 query};
    \node[token] (y2) at (9.4,1.9) {sample\\$y_2$};
    \node[input,minimum width=3.5cm] (cache) at (4.8,.4) {per-layer KV cache\\prompt $\rightarrow$ prompt + $y_1$};
    \draw[arr] (p)--(prefill); \draw[arr] (prefill)--(y1); \draw[arr] (y1)--(decode); \draw[arr] (decode)--(y2);
    \draw[arr,accentvioletdraw] (prefill.south)--(cache.north west);
    \draw[arr,accentvioletdraw] (cache.north east)--(decode.south);
  \end{tikzpicture}
  \par\vspace{6pt}
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
    Forward pass & Input & Cache in & Cache out \\
    \midrule
    prefill & prompt, $[B,P]$ & none & $T=P$ \\
    decode step $i=1,\ldots,n-1$ & $y_i$ alone, $[B,1]$ & $T=P+i-1$ & $T=P+i$ \\
    \bottomrule
  \end{tabular}
  \par\vspace{6pt}
  \raggedright\footnotesize
  Each $y_i$ is selected from the logits of the \textbf{last} position (temperature, top-$k$, sample), exactly as without a cache.\par
  \vspace{3pt}
  $y_n$ is emitted and never fed back: the final cached length is $T=P+n-1$, and it must fit in the context window.
\end{frame}

\begin{frame}{What the cache saves, and what it does not}
  Token-positions pushed through the network to emit $n$ tokens after a prompt of $P$:
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{alertblock}{Without cache}
        \[
          \sum_{i=1}^{n}(P+i-1)\;\approx\;nP+\tfrac{n^2}{2}
        \]
        Every step reprocesses the whole prefix.
      \end{alertblock}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{With cache}
        \[
          P+(n-1)
        \]
        Every token is projected once, in every layer.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{itemize}
      \setlength{\itemsep}{5pt}
    \item \textbf{Not saved:} each decode query still scans all $T$ cached positions.
    \item \textbf{The price:} the reused state must stay resident, per layer and per request.
  \end{itemize}
\end{frame}

\section{Cache shapes, memory and bandwidth}

\begin{frame}{From cache shape to bytes}
  \centering
  \begin{tikzpicture}[x=.7cm,y=.62cm,every node/.style={font=\scriptsize,align=center}]
    \foreach \y/\layer in {3/1,2/2,1/L}{
      \node[anchor=east,font=\small\bfseries] at (-.6,\y+.25) {layer $\layer$};
      \node[anchor=east] at (.5,\y+.25) {$K$}; \node[anchor=east] at (5.3,\y+.25) {$V$};
      \foreach \i in {0,...,4}{
        \fill[themebluebg,draw=themebluedraw] (.65+\i*.72,\y) rectangle (1.30+\i*.72,\y+.5);
        \fill[accentdarkbg,draw=accentdarkdraw] (5.45+\i*.72,\y) rectangle (6.10+\i*.72,\y+.5);
      }
    }
    \node[font=\scriptsize] at (2.4,.3) {cached positions $1\ldots T$};
    \node[font=\scriptsize] at (7.2,.3) {cached positions $1\ldots T$};
  \end{tikzpicture}
  \par\vspace{4pt}
  \small\raggedright
  $L$ layers, batch $B$, $H$ heads of width $d_h$, cached length $T$, $b$ bytes per element:
  \[
    M_{KV}=\underbrace{2}_{K\text{ and }V}\,L\,B\,H\,T\,d_h\,b
  \]
  \begin{block}{Per-token, per-request growth}
    \centering $m_{\text{token}}=2\,L\,H\,d_h\,b$ bytes appended at every decode step
  \end{block}
\end{frame}

\begin{frame}{Worked memory estimate}
  Suppose $L=32$, $H=32$, $d_h=128$, BF16 ($b=2$):
  \[
    m_{\text{token}}=2\cdot32\cdot32\cdot128\cdot2=524{,}288\ \text{bytes}=0.5\ \text{MiB per token}
  \]
  One request at $T=4096$:
  \[
    M_{KV}=4096\cdot0.5\ \text{MiB}=2\ \text{GiB}
  \]
  \begin{alertblock}{Cache memory bounds concurrency}
    Sixteen such requests need 32 GiB of KV state alone, before weights, workspaces, and fragmentation.
  \end{alertblock}
\end{frame}

\begin{frame}{How should the cache grow?}
  \centering
  \begin{tikzpicture}[x=.7cm,y=.7cm,every node/.style={font=\scriptsize,align=center}]
    \node[anchor=east,font=\small\bfseries] at (-.25,3) {concatenate};
    \node[anchor=east,font=\small\bfseries] at (-.25,1.75) {preallocate};
    \node[anchor=east,font=\small\bfseries] at (-.25,.5) {paged};
    \foreach \i in {0,...,3}{\fill[themebluebg,draw=themebluedraw] (\i,2.7) rectangle (\i+.88,3.3);}
    \node[anchor=west] at (4.3,3) {grow by reallocating / copying};
    \foreach \i in {0,...,3}{\fill[themebluebg,draw=themebluedraw] (\i,1.45) rectangle (\i+.88,2.05);}
    \foreach \i in {4,...,7}{\draw[neutrallight,dashed] (\i,1.45) rectangle (\i+.88,2.05);}
    \node[anchor=west] at (8.2,1.75) {reserved};
    \foreach \i in {0,1,3,5}{\fill[accentdarkbg,draw=accentdarkdraw] (\i,.2) rectangle (\i+.88,.8);}
    \node[anchor=west] at (6.2,.5) {map logical to physical blocks};
  \end{tikzpicture}
  \par\vspace{6pt}
  \small Concatenation may repeatedly copy; preallocation reserves capacity; paging grows in blocks and adds indirection and last-block slack.
  \par\footnotesize Module 10 follows multiple requests through a shared paged allocator.
\end{frame}

\begin{frame}{Decode often reads far more bytes than it computes}
  Each new token reads the cached state of every layer, for every active request. Batch-1 decode has little reuse across queries.
  \begin{block}{Bandwidth model}
    \[
      t_{\text{decode}}\gtrsim
      \frac{\text{weight bytes read}+\text{KV bytes read}}{BW_{\text{effective}}}
    \]
  \end{block}
  \begin{alertblock}{Consequence}
    Cache bytes affect capacity and can limit decode speed. The bottleneck depends on context length, batch, model and hardware; precision and layout also matter.
  \end{alertblock}
\end{frame}

\checkpoint{capacity calculation}
{24 GiB is available for cache after weights and workspaces. With the example model at $T=4096$, how many concurrent requests fit with MHA, with GQA at eight KV heads, and with MLA at $d_c=512$, $d_R=64$?}

\begin{frame}{Capacity calculation: one possible answer}
  Only the elements per token and layer change; $L\,T\,b=32\cdot4096\cdot2=262{,}144$ bytes per element.
  \par\vspace{6pt}
  \begin{center}
    \begin{tabular}{@{}lrrr@{}}
      \toprule
      Design & Elements & Per request & Requests in 24 GiB \\
      \midrule
      MHA & 8192 & 2 GiB & 12 \\
      GQA, $H_{kv}=8$ & 2048 & 512 MiB & 48 \\

    \end{tabular}

  \end{itemize}
\end{frame}

\section{Positions, masks and correctness}

\begin{frame}{Positions and mask continue from the cached length}
  \begin{columns}[c]
    \begin{column}{0.60\linewidth}
      \footnotesize
      $T$ cached tokens, $N_{\text{new}}$ new ones, indices from 0:
      \begin{itemize}
          \setlength{\itemsep}{3pt}
        \item \textbf{Positions:} new tokens take $T,\ldots,T+N_{\text{new}}-1$; they never restart at 0.
        \item \textbf{Mask:} rows $T,\ldots,T+N_{\text{new}}-1$ of the causal matrix, all $T+N_{\text{new}}$ columns.
        \item \textbf{One new token:} for an unpadded sequence, its row is all ones; no causal mask is needed.
        \item \textbf{Square causal mask:} correct only when queries and keys start together, that is with an empty cache.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\linewidth}
      \centering
      \begin{tikzpicture}[x=.36cm,y=.36cm,every node/.style={font=\scriptsize}]
        % 7x7 causal matrix; T=4 cached, 3 new queries (rows 4..6)
        \foreach \r in {0,...,6}{\foreach \c in {0,...,6}{
            \pgfmathtruncatemacro{\vis}{\c<=\r ? 1 : 0}
            \pgfmathtruncatemacro{\isnew}{\r>=4 ? 1 : 0}
            \ifnum\isnew=1
              \ifnum\vis=1 \fill[accentdarkdraw,draw=accentdarkdraw] (\c,-\r) rectangle (\c+1,-\r-1);
              \else \fill[white,draw=accentdarkdraw] (\c,-\r) rectangle (\c+1,-\r-1); \fi
            \else
              \ifnum\vis=1 \fill[neutralbg,draw=neutrallight] (\c,-\r) rectangle (\c+1,-\r-1);
              \else \draw[neutrallight] (\c,-\r) rectangle (\c+1,-\r-1); \fi
            \fi
        }}
        \draw[decorate,decoration={brace,amplitude=3pt}] (0,.25)--(4,.25) node[midway,above=3pt]{cached $T$};
        \draw[decorate,decoration={brace,amplitude=3pt}] (4,.25)--(7,.25) node[midway,above=3pt]{new};
        \draw[decorate,decoration={brace,amplitude=3pt,mirror}] (-.25,-4)--(-.25,-7) node[midway,left=3pt]{queries};
        \node[text=neutraldark] at (3.5,-7.6) {keys (columns)};
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \begin{alertblock}{Common failure}
    \small Correct keys and values at wrong positions give plausible but divergent text.
  \end{alertblock}
\end{frame}

\begin{frame}{Full-prefix decoding is the reference oracle}
  For fixed weights, prompt, and token sequence:
  \begin{enumerate}
      \setlength{\itemsep}{5pt}
    \item run a full-prefix forward pass for each step
    \item run cached prefill plus single-token decode
    \item compare next-token logits at every step
    \item feed a prompt in two chunks: a wrong mask offset passes every single-token step and fails here
    \item repeat across batch sizes, prompt lengths, and up to the full context window
  \end{enumerate}
  \begin{block}{Then test generation}
    \small Compare logits within dtype tolerance. Greedy tokens then match, except at near-ties within that tolerance; sampling also needs matching random state.
  \end{block}
\end{frame}

\begin{frame}{Benchmark prefill and decode separately}
  \small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.17\linewidth}>{\raggedright\arraybackslash}p{.43\linewidth}>{\raggedright\arraybackslash}p{.30\linewidth}@{}}
    \toprule
    Phase & Primary metrics & Sweep \\
    \midrule
    prefill & latency, prompt tokens/s, peak memory & prompt length, batch \\
    decode & median interval per token, output tokens/s, cache bytes & cached length, batch \\
    end-to-end & TTFT, total latency & prompt/output distribution \\
    \bottomrule
  \end{tabular}
  \par\vspace{9pt}
  Model TTFT includes prefill and first-token selection. Decode latency can increase with cached length; measure the dependence for the workload.
  \vfill
  \begin{alertblock}{Verify, warm, synchronize}
    Check cached and uncached logits within tolerance before timing. Kernel compilation, allocation, and asynchronous timing otherwise dominate or under-report short decode steps.
  \end{alertblock}
\end{frame}

{Single-token cached decode matches the full-prefix reference. When a prompt is fed in two chunks, logits diverge on the second token of the second chunk. What would you inspect first?}

\begin{frame}{Cache debugging: one possible answer}
  \begin{itemize}
    \item Compare logits at each new position to locate the first mismatch.
    \item Check positions continue from the cached length $T$.
    \item For a chunk of new queries, use causal mask rows starting at $T$: each query sees the cache and earlier positions in the chunk, not later ones.
    \item Inspect cache length and append order; compare layer K/V with a full-prefix run.
  \end{itemize}
\end{frame}

\takeaways
{Causality makes past keys and values exactly reusable: prefill once, then one token per step.}
{Cache bytes grow with layers, requests, cached tokens, per-token elements, and precision; decode is commonly bandwidth-bound.}
{MHA, GQA/MQA, and MLA store $2Hd_h$, $2H_{kv}d_h$, and $d_c+d_R$ elements per token; each is a trained architecture, not a retrofit.}
{Full-prefix logit equivalence is the primary correctness test before benchmarking.}

\begin{frame}{MLA attention: content from the latent, position on the side}
  \footnotesize
  Head $i$, query at $t$, cached position $j$:
  {\small
  \[
    s^{(i)}_{t,j}=
    \underbrace{\big\langle q^C_{t,i},\,W^{UK}_i c_j\big\rangle}_{\text{content}}
    +\underbrace{\big\langle q^R_{t,i},\,k^R_j\big\rangle}_{\text{position}},
    \quad
    o_{t,i}=\sum_{j\le t}\operatorname{softmax}_j\!\Big(\tfrac{s^{(i)}_{t,j}}{\sqrt{d_h+d_R}}\Big)W^{UV}_i c_j
  \]}
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{block}{Why position is separate}
        \footnotesize RoPE on $W^{UK}_ic_j$ would put a rotation that depends on $j$ between the query and $W^{UK}_i$. Rotating only a small key $k^R_j$, shared by all heads, keeps the content path linear.
      \end{block}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{An absorbed decode implementation}
        \footnotesize
        $\langle q,W^{UK}_ic_j\rangle=\langle W^{UK\top}_i q,\,c_j\rangle$: project the one query down and score the latents directly. $W^{UV}_i$ folds into the output projection the same way.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \footnotesize The query has a content part $q^C$ and a RoPE part $q^R$ per head; queries are never cached.
\end{frame}

\checkpoint{cache debugging}
{Cached logits match for short prompts but diverge immediately after a page boundary. Which state and indexing checks isolate layout bugs from positional-encoding bugs?}

\begin{frame}{Cache debugging: one possible answer}
  \small
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item \textbf{Localize:} compare cached and full-prefix logits step by step; note the first divergent step and layer.
    \item \textbf{Remove the layout:} rerun the same prompt with a contiguous cache, or a block larger than the sequence. If logits now match, investigate the paged path, including its position and mask handling.
    \item \textbf{State check:} rebuild $K,V$ in logical order through the block table and compare them with the full-prefix $K,V$. Inspect the new table entry, the in-block offset, the per-request length and the mask at the boundary.
    \item \textbf{Position check:} also verify that the new token's position $t$ continues across the boundary instead of restarting inside the new block.
  \end{itemize}
\end{frame}
