    \node[attn] (attention) at (8.55,2.35) {attention\\$q_t$ with $K,V$};
    \node[proj] (out) at (10.9,2.35) {output\\$o_t$};
    \draw[arrow] (xt)--(qt); \draw[arrow] (xt)--(kvt);
    \draw[arrow] (kvt)--(cache);
    \node[font=\scriptsize,text=accentdark] at (3.56,.3) {append};
    \draw[arrow] (qt)--(attention);
    \draw[arrow,accentdarkdraw] (cache.north) -- ++(0,.35) -| (attention.south);
    \draw[arrow] (attention)--(out);
  \end{tikzpicture}
  \par\vspace{4pt}
  \[
    o_t=\operatorname{softmax}\!\left(q_tK_{1:t}^{\top}/\sqrt{d_h}\right)V_{1:t}
  \]
  \footnotesize\raggedright Per layer, the cache is a pair $K,V$ of shape $[B,H,T,d_h]$, here with $T=t-1$. Only the new token is projected: its $k_t,v_t$ of shape $[B,H,1,d_h]$ are concatenated along the $T$ axis, and its query scans all $t$ positions.
\end{frame}

\begin{frame}{Generating $n$ tokens: one prefill, $n-1$ decode steps}
  \small
  \begin{center}
    \begin{tabular}{@{}llll@{}}
      \toprule
      Forward pass & Input & Cache in & Cache out \\
      \midrule
      prefill & prompt, $[B,P]$ & none & $T=P$ \\
      decode step $i=1,\ldots,n-1$ & $y_i$ alone, $[B,1]$ & $T=P+i-1$ & $T=P+i$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{enumerate}
      \setlength{\itemsep}{4pt}
    \item Prefill the whole prompt once and keep the $L$ per-layer $(K,V)$ pairs.
    \item Select $y_i$ from the logits of the \textbf{last} position: temperature, top-$k$, then sample (argmax at temperature 0), exactly as without a cache.
    \item Feed $y_i$ alone with the cache to obtain the logits for $y_{i+1}$.
  \end{enumerate}
  \begin{block}{The last token is never fed back}
    $y_n$ is emitted without a forward pass, so the final cached length is $T=P+n-1$, and this must fit in the context window.
  \end{block}
\end{frame}

\begin{frame}{The cache changes asymptotic projection work}
  For $n$ generated tokens after a prompt:
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      \begin{alertblock}{Without cache}
        Reproject the growing prefix at every step. Prefix projection work accumulates roughly quadratically in generated length.
      \end{alertblock}
    \end{column}
    \begin{column}{0.47\linewidth}
      \begin{block}{With cache}
        Project each token's key and value once. Attention still scans a context that grows linearly per decode step.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \centering The cache removes redundant projection work. It does not make full-context attention constant-time.
\end{frame}

\section{Cache memory and bandwidth}

\begin{frame}{A separate cache for every layer and request}
  \centering
  \begin{tikzpicture}[x=.7cm,y=.7cm,every node/.style={font=\scriptsize,align=center}]
    \foreach \y/\layer in {3/1,2/2,1/L}{
      \node[anchor=east,font=\small\bfseries] at (-.6,\y+.25) {layer $\layer$};
      \node[anchor=east] at (.5,\y+.25) {$K$}; \node[anchor=east] at (5.3,\y+.25) {$V$};
      \foreach \i in {0,...,4}{
        \fill[themebluebg,draw=themebluedraw] (.65+\i*.72,\y) rectangle (1.30+\i*.72,\y+.5);
        \fill[accentdarkbg,draw=accentdarkdraw] (5.45+\i*.72,\y) rectangle (6.10+\i*.72,\y+.5);
      }
    }
    \node[font=\scriptsize] at (2.4,.3) {cached positions $1\ldots T$};
    \node[font=\scriptsize] at (7.2,.3) {cached positions $1\ldots T$};
  \end{tikzpicture}
  \par\vspace{5pt}
  \small Every new token appends one K and one V vector \emph{per layer}; each vector has $H_{kv}d_h$ elements.
  \begin{block}{Serving implication}
    More active requests or longer contexts mean more resident cache state. We count its bytes next.
  \end{block}
\end{frame}

\begin{frame}{Cache shape determines capacity}
  For $L$ layers, batch $B$, $H_{kv}$ key-value heads, cached length $T$, head width $d_h$, and $b$ bytes per element:
  \[
    M_{KV}=2\,L\,B\,H_{kv}\,T\,d_h\,b
  \]
  The factor 2 stores both keys and values. After a prompt of $P$ tokens and $n$ emitted tokens, $T=P+n-1$.
  \begin{block}{Per-token, per-request growth}
    \[
      m_{\text{token}}=2L H_{kv}d_hb
    \]
    This number links model architecture directly to serving concurrency.
  \end{block}
\end{frame}

\begin{frame}{Worked memory estimate}
  Suppose $L=32$, $H_{kv}=32$, $d_h=128$, BF16, and $T=4096$:
  \[
    M_{KV}=2\cdot32\cdot1\cdot32\cdot4096\cdot128\cdot2
    \approx 2\ \text{GiB}
  \]
  \begin{alertblock}{Serving implication}
    Sixteen concurrent requests at this cached length would need roughly 32 GiB for KV state alone, before weights, workspaces, and fragmentation.
  \end{alertblock}
\end{frame}

\begin{frame}{Decode often reads far more bytes than it computes}
  Each new token reads the cached $K,V$ for every active layer and request. Batch-1 decode has little reuse across queries.
  \begin{block}{Bandwidth model}
    \[
      t_{\text{decode}}\gtrsim
      \frac{\text{weight bytes read}+\text{KV bytes read}}{BW_{\text{effective}}}
    \]
  \end{block}
  \begin{alertblock}{Consequence}
    Reducing KV heads, using lower cache precision, batching requests, and improving memory layout can matter more than reducing arithmetic.
  \end{alertblock}
\end{frame}

\checkpoint{capacity calculation}
{Recompute the worked example with eight KV heads. How many 4096-token requests fit in 32 GiB if 24 GiB is available for cache after weights and workspaces?}

\begin{frame}{Capacity calculation: one possible answer}
  Only $H_{kv}$ changes, from 32 to 8:
  \[
    M_{KV}=2\cdot32\cdot1\cdot8\cdot4096\cdot128\cdot2
    =536{,}870{,}912\ \text{bytes}=0.5\ \text{GiB per request}
  \]
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item A quarter of the KV heads gives a quarter of the 2 GiB in the worked example.
    \item Capacity: $24\ \text{GiB}/0.5\ \text{GiB}=48$ concurrent requests.
    \item With 32 KV heads the same budget holds only $24/2=12$ requests.
    \item This is an upper bound: block slack and fragmentation reduce it.
  \end{itemize}
\end{frame}

\section{Correctness and benchmarking}

\begin{frame}{Positions and mask continue from the cached length}
  \begin{columns}[c]
    \begin{column}{0.60\linewidth}
      \footnotesize
      $T$ cached tokens, $N_{\text{new}}$ new ones, indices from 0:
      \begin{itemize}
          \setlength{\itemsep}{3pt}
        \item \textbf{Positions:} new tokens take $T,\ldots,T+N_{\text{new}}-1$; they never restart at 0.
        \item \textbf{Mask:} rows $T,\ldots,T+N_{\text{new}}-1$ of the causal matrix, all $T+N_{\text{new}}$ columns.
        \item \textbf{One new token:} its row is all ones; no mask is needed.
        \item \textbf{Square causal mask:} correct only when queries and keys start together, that is with an empty cache.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\linewidth}
      \centering
      \begin{tikzpicture}[x=.36cm,y=.36cm,every node/.style={font=\scriptsize}]

\begin{frame}{Full-prefix decoding is the reference oracle}
  For fixed weights, prompt, and token sequence:
  \begin{enumerate}
      \setlength{\itemsep}{5pt}
    \item run a full-prefix forward pass for each step
    \item run cached prefill plus single-token decode
    \item compare next-token logits at every step
    \item feed a prompt in two chunks: a wrong mask offset passes every single-token step and fails here
    \item repeat across batch sizes, prompt lengths, and up to the full context window
  \end{enumerate}
  \begin{block}{Then test generation}
    \small Compare logits within dtype tolerance. Greedy tokens then match, except at near-ties within that tolerance; sampling also needs matching random state.
  \end{block}
\end{frame}

\begin{frame}{Benchmark prefill and decode separately}
  \small
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{.17\linewidth}>{\raggedright\arraybackslash}p{.43\linewidth}>{\raggedright\arraybackslash}p{.30\linewidth}@{}}
    \toprule
    Phase & Primary metrics & Sweep \\
    \midrule
    prefill & latency, prompt tokens/s, peak memory & prompt length, batch \\
    decode & median interval per token, output tokens/s, cache bytes & cached length, batch \\
    end-to-end & TTFT, total latency & prompt/output distribution \\
    \bottomrule
  \end{tabular}
  \par\vspace{9pt}
  Model TTFT $=$ prefill $+$ first token selection: it grows with $P$. The decode interval grows only slowly with $T$.
  \vfill
  \begin{alertblock}{Verify, warm, synchronize}
    Time only after cached and uncached tokens match. Kernel compilation, allocation, and asynchronous timing otherwise dominate or under-report short decode steps.
  \end{alertblock}
\end{frame}

\checkpoint{cache debugging}
{Cached logits match for short prompts but diverge immediately after a page boundary. Which state and indexing checks isolate layout bugs from positional-encoding bugs?}

\begin{frame}{Cache debugging: one possible answer}
  \small
  \begin{itemize}
      \setlength{\itemsep}{8pt}
    \item \textbf{Localize:} compare cached and full-prefix logits step by step; note the first divergent step and layer.
    \item \textbf{Remove the layout:} rerun the same prompt with a contiguous cache, or a block larger than the sequence. If the logits now match, positions are correct and the paged layout is at fault.
    \item \textbf{State check:} rebuild $K,V$ in logical order through the block table and compare them with the full-prefix $K,V$. Inspect the new table entry, the in-block offset, the per-request length and the mask at the boundary.
    \item \textbf{Position check:} if the gathered $K,V$ match but logits differ, verify that the new token's position $t$ continues across the boundary instead of restarting inside the new block.
  \end{itemize}
\end{frame}

\takeaways
{KV caching reuses past key and value projections exactly.}
{Cache memory grows with layers, KV heads, requests, cached tokens, head width, and precision.}
{Decode is commonly bandwidth-bound, so cache architecture and layout shape performance.}
{Full-prefix logit equivalence is the primary correctness test before benchmarking.}
