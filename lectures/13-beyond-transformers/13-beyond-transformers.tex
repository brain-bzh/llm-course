%================= SECTION 1 ==========================
\section{Architectural Limits of Dense Attention}

\begin{frame}{The Scaling Frontier: Sparsity \& Alternative Attention}
  \begin{columns}[c]
    \begin{column}{0.58\linewidth}
      \small
      \begin{itemize}
        \setlength{\itemsep}{5pt}
        \item \textbf{Dense Transformer constraints:}\; Every parameter is activated for every token ($\mathcal{O}(N)$ compute cost), and KV-cache footprint grows linearly with context length $T$.
        \item \textbf{Sparse Mixture-of-Experts (MoE):}\; Route each token to Top-$k$ of $E$ specialized MLP experts. Dramatically expands parameter capacity without increasing active FLOPs.
        \item \textbf{Recurrence \& State Space Models:}\; Mamba / RWKV replace softmax attention with linear recurrent states, achieving $\mathcal{O}(1)$ inference memory per token.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\linewidth}
      \centering
      \begin{tikzpicture}[
        every node/.style={font=\scriptsize},
